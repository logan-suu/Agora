// Real Seatbelt payloads and kernel audit identities; no test doubles.
import { expect, it } from 'vitest';
import { probeLocalCommandStop } from './local-command-stop-fixture';

it('stops only the registered process birth identity and escalates ignored TERM', async () => {
  const result = await probeLocalCommandStop('ignore-term');
  expect(result.forgedIdentityState).toBe('identityMismatch');
  expect(result.aliveAfterForgedSignal).toBe(true);
  expect(result.stop?.registeredState).toBe('stopped');
  expect(result.stop?.signals).toEqual(['TERM', 'KILL']);
  expect(result.stop?.assurance).toBe('registered-only');
  expect(result.stop?.durationMs).toBeLessThan(5000);
  expect(result.output.mainResult.signal).toBe('SIGKILL');
  expect(result.output.stdout.truncated).toBe(false);
}, 20_000);

it('caps retained output without treating a successful exit as trusted completion', async () => {
  const result = await probeLocalCommandStop('flood');
  expect(result.output.mainResult.exitCode).toBe(0);
  expect(result.output.stdout.bytes.length).toBe(1024 * 1024);
  expect(result.output.stderr.bytes.length).toBe(1024 * 1024);
  expect(result.output.stdout.truncated).toBe(true);
  expect(result.output.stderr.truncated).toBe(true);
  expect(result.output.stdout.observedBytes).toBeGreaterThan(1024 * 1024);
  expect(result.output.stdout.error).toBeNull();
}, 20_000);

it('closes inherited pipes within the tail deadline while exposing incomplete output', async () => {
  const result = await probeLocalCommandStop('inherited-pipe');
  expect(result.output.mainResult.exitCode).toBe(0);
  expect(result.output.stdout.truncated).toBe(true);
  expect(result.output.stderr.truncated).toBe(true);
  expect(result.output.tailDurationMs).toBeGreaterThanOrEqual(900);
  expect(result.output.tailDurationMs).toBeLessThan(2500);
}, 20_000);

it('finishes a TERM-responsive registered process without sending KILL', async () => {
  const result = await probeLocalCommandStop('normal-term');
  expect(result.stop?.registeredState).toBe('stopped');
  expect(result.stop?.signals).toEqual(['TERM']);
  expect(result.output.mainResult.signal).toBe('SIGTERM');
}, 20_000);

it('returns needsAttention for a mismatched registered birth without sending a signal', async () => {
  const result = await probeLocalCommandStop('wrong-birth');
  expect(result.stop?.registeredState).toBe('needsAttention');
  expect(result.stop?.signals).toEqual([]);
  expect(result.stop?.states).toEqual(['identityMismatch']);
  expect(result.aliveAfterForgedSignal).toBe(true);
}, 20_000);
