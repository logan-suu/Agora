/** Internal launch gate, not a workspace authorization or product entry point.
 * A trusted caller must bind binaries, policy, roots and argv to a current grant.
 * Only this controller can release its suspended, durably registered executable.
 */
import { type ChildProcess, spawn } from 'node:child_process';
import { createHash } from 'node:crypto';
import { performance } from 'node:perf_hooks';
import type { Duplex } from 'node:stream';
import { setImmediate as nextTurn } from 'node:timers/promises';
import { LocalCommandBinding } from './local-command-binding';
import type { LocalCommandJournal } from './local-command-journal';
import {
  captureLocalProcess,
  discoverLocalProcessCohort,
  inspectLocalProcess,
  type LocalProcessIdentity,
  prepareLocalProcessControl,
  stopRegisteredLocalProcesses,
} from './local-command-stop';
import { superviseLocalCommand } from './local-command-supervisor';

type Options = {
  bootstrap: string;
  executable: string;
  argv: string[];
  policy: string;
  outputRoot: string;
  helper: string;
  journal: LocalCommandJournal;
  commandId: string;
  revision: number;
  binding: LocalCommandBinding;
  /** Synchronous trusted control-plane check; never supplied by project code. */
  authorize: (stage: 'admission' | 'spawn' | 'register' | 'release') => boolean;
  signal?: AbortSignal;
};

type CaptureAttempt = {
  attempt: number;
  durationMs: number;
  outcome: 'captured' | 'timed_out' | 'failed';
};

/** Only the private, ready bootstrap is eligible. No START has been sent yet.
 * Retry read-only timeouts, never a signal, invalid identity or authorization failure. */
async function captureBootstrap(
  child: ChildProcess,
  helper: string,
  deadline: number,
  authorize: () => void,
  admit: () => boolean,
  confirmAlive: () => Promise<void>,
  attempts: CaptureAttempt[],
): Promise<LocalProcessIdentity> {
  const check = () => {
    authorize();
    if (child.exitCode !== null || child.signalCode !== null || child.killed)
      throw new Error('command_bootstrap_failed');
    if (performance.now() >= deadline) throw new Error('command_capture_deadline');
  };
  for (let attempt = 1; attempt <= 3; attempt++) {
    check();
    if (deadline - performance.now() < 250) throw new Error('command_capture_deadline');
    const started = performance.now();
    let identity: LocalProcessIdentity | undefined;
    let outcome: CaptureAttempt['outcome'] = 'captured';
    try {
      identity = captureLocalProcess(helper, child.pid as number, admit);
    } catch (caught) {
      outcome = (caught as NodeJS.ErrnoException)?.code === 'ETIMEDOUT' ? 'timed_out' : 'failed';
    }
    attempts.push({ attempt, durationMs: performance.now() - started, outcome });
    // Deliver exit/cancellation events before checking the same owned child again.
    // A stale numeric PID alone must never become retry authority.
    await nextTurn();
    check();
    // ChildProcess exit notification can lag even after one event-loop turn.
    // Require a fresh reply on the original private endpoint before accepting
    // a token or trying the PID again; no project child exists at this stage.
    await confirmAlive();
    check();
    if (identity) return identity;
    if (outcome !== 'timed_out') throw new Error('command_capture_failed');
  }
  throw new Error('command_capture_timeout');
}

function protocol(channel: Duplex, child: ChildProcess, deadline: number) {
  let buffer = '';
  let bytes = 0;
  let closed = false;
  let failure = false;
  let wake: (() => void) | undefined;
  const data = (chunk: Buffer) => {
    bytes += chunk.length;
    if (bytes > 256) failure = true;
    else buffer += chunk.toString('ascii');
    wake?.();
  };
  const end = () => {
    closed = true;
    wake?.();
  };
  const error = () => {
    failure = true;
    wake?.();
  };
  channel.on('data', data);
  channel.on('end', end);
  channel.on('error', error);
  child.on('error', error);
  return {
    async line(until = deadline): Promise<string> {
      for (;;) {
        if (failure) throw new Error('command_bootstrap_failed');
        const newline = buffer.indexOf('\n');
        if (newline >= 0) {
          const result = buffer.slice(0, newline);
          buffer = buffer.slice(newline + 1);
          return result;
        }
        if (closed || performance.now() >= until) throw new Error('command_bootstrap_failed');
        await new Promise<void>((resolve) => {
          const timer = setTimeout(resolve, Math.max(1, until - performance.now()));
          wake = () => {
            clearTimeout(timer);
            resolve();
          };
        });
        wake = undefined;
      }
    },
    close() {
      channel.off('data', data);
      channel.off('end', end);
      channel.off('error', error);
      child.off('error', error);
      channel.destroy();
    },
  };
}

export async function runHeldLocalCommand(options: Options) {
  options = { ...options, argv: [...options.argv] };
  let released = false;
  let launchDurable = false;
  let targetIdentity: LocalProcessIdentity | null = null;
  let error: string | null = null;
  let payloadResult: { exitCode: number | null; signal: number | null } | null = null;
  let observation: Awaited<ReturnType<typeof superviseLocalCommand>> | null = null;
  let startupStop: Awaited<ReturnType<typeof stopRegisteredLocalProcesses>> | null = null;
  let reader: ReturnType<typeof protocol> | undefined;
  let child: ReturnType<typeof spawn> | undefined;
  const known: LocalProcessIdentity[] = [];
  const captureAttempts: CaptureAttempt[] = [];
  let controlReady: { durationMs: number; outcome: 'ready' | 'failed' } | null = null;
  let pending: Promise<Awaited<ReturnType<typeof superviseLocalCommand>>> | undefined;
  const cancel = new AbortController();
  const aborted = () => cancel.abort();
  options.signal?.addEventListener('abort', aborted, { once: true });
  const authorized = (stage: Parameters<Options['authorize']>[0]) => {
    if (options.binding.check(stage)) throw new Error('command_binding_invalidated');
    if (options.signal?.aborted || !options.authorize(stage))
      throw new Error('command_start_denied');
    if (options.binding.check(stage)) throw new Error('command_binding_invalidated');
  };
  try {
    let current = options.journal.snapshot().records.find((r) => r.commandId === options.commandId);
    if (
      process.platform !== 'darwin' ||
      !current ||
      current.revision !== options.revision ||
      current.resourceState !== 'reserved' ||
      !(options.binding instanceof LocalCommandBinding) ||
      !options.binding.matches(options, current.workspaceId) ||
      current.birthIdentities.length !== 0 ||
      current.policyHash !== createHash('sha256').update(options.policy).digest('hex') ||
      !current.roots.some((root) => root.path === options.outputRoot)
    )
      throw new Error('invalid_command_start');
    authorized('admission');
    current = options.journal.bindCommand(
      current.commandId,
      current.revision,
      options.binding.snapshot(),
    );
    const startedAt = performance.now();
    const deadline = startedAt + 5000;
    try {
      prepareLocalProcessControl(options.helper, deadline, () => options.binding.controlAllowed());
      controlReady = { durationMs: performance.now() - startedAt, outcome: 'ready' };
    } catch (caught) {
      controlReady = { durationMs: performance.now() - startedAt, outcome: 'failed' };
      throw caught;
    }
    authorized('admission');
    if (performance.now() >= deadline) throw new Error('command_capture_deadline');
    child = spawn(
      '/usr/bin/sandbox-exec',
      ['-p', options.policy, options.bootstrap, options.executable, ...options.argv],
      {
        cwd: options.outputRoot,
        // No inherited keys, runtime injection variables, shell configuration or user HOME.
        env: {
          NODE_ENV: 'production',
          HOME: options.outputRoot,
          TMPDIR: options.outputRoot,
          LANG: 'C',
          LC_ALL: 'C',
        },
        stdio: ['ignore', 'pipe', 'pipe', 'pipe'],
      },
    );
    const channel = child.stdio[3] as Duplex;
    reader = protocol(channel, child, deadline);
    if ((await reader.line()) !== 'ready') throw new Error('command_bootstrap_failed');
    const bootstrapIdentity = await captureBootstrap(
      child,
      options.helper,
      deadline,
      () => authorized('spawn'),
      () => options.binding.controlAllowed(),
      async () => {
        channel.write('p');
        if ((await reader?.line()) !== 'ready') throw new Error('command_bootstrap_failed');
      },
      captureAttempts,
    );
    known.push(bootstrapIdentity);
    current = options.journal.registerBirth(current.commandId, current.revision, bootstrapIdentity);
    authorized('spawn');
    channel.write('s');
    const line = await reader.line();
    if (!/^child:[1-9][0-9]{0,9}$/.test(line)) throw new Error('command_bootstrap_failed');
    const pid = Number(line.slice(6));
    // The private bootstrap nominates a PID; the kernel must prove its parent relation.
    const discovered = discoverLocalProcessCohort(
      options.helper,
      [bootstrapIdentity],
      deadline,
      () => options.binding.controlAllowed(),
    );
    for (const identity of discovered.identities.slice(1)) known.push(identity);
    const relation = discovered.relations.find(
      (r) => r.child[5] === pid && r.parent.join(':') === bootstrapIdentity.join(':'),
    );
    if (
      discovered.observationState !== 'observed' ||
      discovered.relations.length !== 1 ||
      !relation
    )
      throw new Error('command_birth_unproven');
    authorized('register');
    current = options.journal.registerBirth(
      current.commandId,
      current.revision,
      relation.child,
      relation.parent,
    );
    targetIdentity = [...relation.child];
    const releaseRevision = current.revision;
    pending = superviseLocalCommand({
      child,
      identity: bootstrapIdentity,
      helper: options.helper,
      journal: options.journal,
      commandId: current.commandId,
      revision: current.revision,
      startedAt,
      signal: cancel.signal,
      binding: options.binding,
      release() {
        try {
          authorized('release');
          // The callback may revoke durable state; re-read immediately before release.
          const active = options.journal
            .snapshot()
            .records.find((r) => r.commandId === options.commandId);
          if (
            !active ||
            active.revision !== releaseRevision ||
            active.resourceState !== 'reserved' ||
            performance.now() >= deadline
          )
            throw new Error('command_start_denied');
          const resumed = inspectLocalProcess(options.helper, relation.child, 'CONT', 250, () =>
            options.binding.controlAllowed(),
          );
          if (!resumed.signalSent) throw new Error('command_release_failed');
          released = true;
          return true;
        } catch (caught) {
          error = caught instanceof Error ? caught.message : 'command_release_failed';
          return false;
        }
      },
    });
    observation = await pending;
    if (observation.bindingFailure) error = 'command_binding_invalidated';
    if (released) {
      const result = await reader.line(performance.now() + 250);
      const match = /^(exit|signal):([0-9]{1,3})$/.exec(result);
      if (!match || Number(match[2]) > 255 || (match[1] === 'signal' && Number(match[2]) === 0))
        throw new Error('command_bootstrap_failed');
      payloadResult =
        match[1] === 'exit'
          ? { exitCode: Number(match[2]), signal: null }
          : { exitCode: null, signal: Number(match[2]) };
    }
  } catch (caught) {
    error ??= caught instanceof Error ? caught.message : 'command_start_failed';
    cancel.abort();
    if (pending) {
      try {
        observation = await pending;
      } catch {
        error = 'command_supervision_failed';
      }
    }
  } finally {
    if (!observation && known.length)
      startupStop = await stopRegisteredLocalProcesses(options.helper, known, undefined, () =>
        options.binding.controlAllowed(),
      );
    try {
      const active = options.journal
        .snapshot()
        .records.find((r) => r.commandId === options.commandId);
      if (active?.resourceState === 'reserved')
        options.journal.quarantine(active.commandId, active.revision, 'control_failure');
    } catch {
      // An unreadable/locked reservation stays blocking; never reset or release it.
      error ??= 'command_journal_unavailable';
    }
    try {
      const active = options.journal
        .snapshot()
        .records.find((r) => r.commandId === options.commandId);
      if (active) {
        options.journal.recordLaunch(active.commandId, active.revision, {
          version: 1,
          targetIdentity,
          released,
          payloadResult,
          error:
            error === null
              ? null
              : /^[a-z][a-z0-9_]{0,127}$/.test(error)
                ? error
                : 'command_start_failed',
        });
        launchDurable = true;
      } else error ??= 'command_launch_not_durable';
    } catch {
      // Preserve in-memory process facts, but never claim persistence on uncertain I/O.
      error ??= 'command_launch_not_durable';
    }
    reader?.close();
    child?.stdout?.destroy();
    child?.stderr?.destroy();
    options.signal?.removeEventListener('abort', aborted);
  }
  return {
    released,
    payloadResult,
    observation,
    startupStop,
    error,
    launchDurable,
    captureAttempts,
    controlReady,
  };
}
