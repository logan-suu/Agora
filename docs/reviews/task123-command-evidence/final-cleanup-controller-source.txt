}
function same(p, s) {
  const now = fs.lstatSync(p);
  return now.isDirectory() && !now.isSymbolicLink() && now.uid === s.uid && now.dev === s.dev && now.ino === s.ino && fs.realpathSync(p) === p;
}
function manifest(root) {
  const entries = [];
  const visit = (directory, relative = '') => {
    for (const name of fs.readdirSync(directory).sort()) {
      const full = path.join(directory, name), rel = path.join(relative, name), s = fs.lstatSync(full);
      const entry = { path: rel, mode: s.mode, size: s.size, dev: s.dev, ino: s.ino };
      if (s.isSymbolicLink()) entries.push({ ...entry, kind: 'symlink', target: fs.readlinkSync(full) });
      else if (s.isDirectory()) { entries.push({ ...entry, kind: 'directory' }); visit(full, rel); }
      else if (s.isFile()) entries.push({ ...entry, kind: 'file', sha256: hash(fs.readFileSync(full)) });
      else entries.push({ ...entry, kind: 'special' });
    }
  };
  visit(root);
  return { entries, sha256: hash(JSON.stringify(entries)), logicalBytes: entries.filter(e => e.kind === 'file').reduce((n,e) => n + e.size, 0) };
}
