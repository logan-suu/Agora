/* Fixed Seatbelt payload and trusted audit-token inspector; not a production supervisor. */
#define _DARWIN_C_SOURCE
#include <bsm/audit.h>
#include <errno.h>
#include <libproc.h>
#include <mach/mach.h>
#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/wait.h>
#include <unistd.h>

static int token_for_pid(int pid, audit_token_t *token) {
  mach_port_t name = MACH_PORT_NULL;
  kern_return_t result = task_name_for_pid(mach_task_self(), pid, &name);
  if (result != KERN_SUCCESS) return result;
  mach_msg_type_number_t count = TASK_AUDIT_TOKEN_COUNT;
  result = task_info(name, TASK_AUDIT_TOKEN, (task_info_t)token, &count);
  mach_port_deallocate(mach_task_self(), name);
  return result;
}
static void report(const char *stage) {
  auditinfo_addr_t info = {0};
  int result = getaudit_addr(&info, sizeof(info));
  printf("{\"stage\":\"%s\",\"pid\":%d,\"ppid\":%d,\"pgid\":%d,\"sid\":%d,\"auditResult\":%d,\"asid\":%u}\n",
         stage, getpid(), getppid(), getpgrp(), getsid(0), result, info.ai_asid);
  fflush(stdout);
}
int main(int argc, char **argv) {
  alarm(8);
  if (argc == 3 && !strcmp(argv[1], "scan")) {
    unsigned int asid = (unsigned int)strtoul(argv[2], NULL, 10);
    const int capacity = 65536;
    pid_t *pids = calloc((size_t)capacity, sizeof(pid_t));
    if (!pids) return 74;
    int bytes = proc_listpids(PROC_UID_ONLY, getuid(), pids, capacity * (int)sizeof(pid_t));
    if (bytes <= 0 || bytes >= capacity * (int)sizeof(pid_t) || bytes % sizeof(pid_t)) return 75;
    int unresolved = 0, members = 0;
    for (int i = 0; i < bytes / (int)sizeof(pid_t); i++) {
      if (pids[i] <= 0) continue;
      audit_token_t token = {{0}};
      int result = token_for_pid(pids[i], &token);
      if (result != KERN_SUCCESS || token.val[5] != (unsigned)pids[i] || token.val[1] != getuid()) {
        unresolved++;
        continue;
      }
      if (token.val[6] != asid) continue;
      members++;
      printf("{\"stage\":\"member\",\"pid\":%d,\"asid\":%u,\"pidVersion\":%u}\n", pids[i], asid, token.val[7]);
    }
    printf("{\"stage\":\"scan\",\"members\":%d,\"unresolved\":%d,\"examined\":%d}\n", members, unresolved, bytes / (int)sizeof(pid_t));
    free(pids);
    return 0;
  }
  if (argc == 4 && !strcmp(argv[1], "stop")) {
    int pid = atoi(argv[2]);
    unsigned int asid = (unsigned int)strtoul(argv[3], NULL, 10);
    audit_token_t token = {{0}};
    int info_result = token_for_pid(pid, &token);
    if (info_result != KERN_SUCCESS || token.val[6] != asid || token.val[1] != getuid() || token.val[5] != (unsigned)pid) return 65;
    audit_token_t forged = token;
    forged.val[7]++;
    errno = 0;
    int wrong = proc_signal_with_audittoken(&forged, SIGTERM);
    int wrong_error = errno;
    audit_token_t after = {{0}};
    int alive = token_for_pid(pid, &after) == KERN_SUCCESS && !memcmp(&after, &token, sizeof(token));
    errno = 0;
    int correct = proc_signal_with_audittoken(&token, SIGTERM);
    int correct_error = errno;
    printf("{\"stage\":\"stop\",\"pid\":%d,\"asid\":%u,\"pidVersion\":%u,\"wrongResult\":%d,\"wrongErrno\":%d,\"aliveAfterWrong\":%s,\"correctResult\":%d,\"correctErrno\":%d}\n",
           pid, asid, token.val[7], wrong, wrong_error, alive ? "true" : "false", correct, correct_error);
    return 0;
  }
  if (argc != 2 || strcmp(argv[1], "escape")) return 64;
  int ready[2];
  if (pipe(ready)) return 66;
  report("parent");
  pid_t child = fork();
  if (child < 0) return 67;
  if (child == 0) {
    close(ready[0]);
    alarm(6);
    if (setsid() < 0) _exit(68);
    pid_t grandchild = fork();
    if (grandchild < 0) _exit(69);
    if (grandchild == 0) {
      alarm(5);
      if (setpgid(0, 0) < 0) _exit(70);
      report("grandchild");
      if (write(ready[1], "x", 1) != 1) _exit(71);
      close(ready[1]);
      sleep(4);
      report("self-exit");
      _exit(0);
    }
    report("intermediate-exit");
    _exit(0);
  }
  close(ready[1]);
  char byte;
  if (read(ready[0], &byte, 1) != 1) return 72;
  close(ready[0]);
  int status;
  if (waitpid(child, &status, 0) != child || status != 0) return 73;
  report("parent-exit");
  return 0;
}
