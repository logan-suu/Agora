/* Fixed feasibility probe; never accepts user code or leaves live descendants. */
#define _DARWIN_C_SOURCE
#include <errno.h>
#include <spawn.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/wait.h>
#include <unistd.h>
extern char **environ;

static void report(const char *mode, int result, int error) {
  printf("{\"mode\":\"%s\",\"result\":%d,\"errno\":%d,\"pid\":%d,\"pgid\":%d,\"sid\":%d}\n",
         mode, result, error, getpid(), getpgrp(), getsid(0));
  fflush(stdout);
}

int main(int argc, char **argv) {
  if (argc != 2) return 64;
  alarm(5);
  if (!strcmp(argv[1], "child")) { report("spawned", 0, 0); return 0; }
  report("initial", 0, 0);
  if (!strcmp(argv[1], "setsid") || !strcmp(argv[1], "setpgid")) {
    pid_t child = fork();
    if (child < 0) return 65;
    if (child == 0) {
      errno = 0;
      int result = !strcmp(argv[1], "setsid") ? setsid() : setpgid(0, 0);
      report(argv[1], result, result < 0 ? errno : 0);
      _exit(0);
    }
    int status;
    if (waitpid(child, &status, 0) != child) return 66;
    report("waited", status, 0);
    return 0;
  }
  posix_spawnattr_t attr;
  if (posix_spawnattr_init(&attr)) return 67;
  short flags = !strcmp(argv[1], "spawn-setsid") ? POSIX_SPAWN_SETSID :
                !strcmp(argv[1], "spawn-setpgid") ? POSIX_SPAWN_SETPGROUP : 0;
  if (posix_spawnattr_setflags(&attr, flags) || posix_spawnattr_setpgroup(&attr, 0)) return 68;
  pid_t child = -1;
  char *args[] = {argv[0], "child", NULL};
  int error = posix_spawn(&child, argv[0], NULL, &attr, args, environ);
  report(argv[1], error ? -1 : child, error);
  posix_spawnattr_destroy(&attr);
  if (!error) {
    int status;
    if (waitpid(child, &status, 0) != child) return 69;
    report("waited", status, 0);
  }
  return 0;
}
