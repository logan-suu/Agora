/* Trusted Seatbelt bootstrap. Never accepts project input on the control fd.
 * The controller persists kernel identities before sending START/SIGCONT.
 * No project code runs before POSIX_SPAWN_START_SUSPENDED is released. */
#define _DARWIN_C_SOURCE
#include <errno.h>
#include <spawn.h>
#include <stdio.h>
#include <string.h>
#include <sys/wait.h>
#include <unistd.h>

extern char **environ;

int main(int argc, char **argv) {
  if (argc < 2 || argv[1][0] != '/') return 64;
  if (dprintf(3, "ready\n") < 0) return 65;
  char command = 0;
  ssize_t count;
  do { count = read(3, &command, 1); } while (count < 0 && errno == EINTR);
  if (count != 1 || command != 's') return 66;
  posix_spawnattr_t attr;
  posix_spawn_file_actions_t actions;
  if (posix_spawnattr_init(&attr) || posix_spawn_file_actions_init(&actions)) return 67;
  int error = posix_spawnattr_setflags(&attr,
    POSIX_SPAWN_START_SUSPENDED | POSIX_SPAWN_CLOEXEC_DEFAULT);
  for (int fd = 0; fd < 3 && !error; fd++)
    error = posix_spawn_file_actions_adddup2(&actions, fd, fd);
  pid_t child = 0;
  if (!error) error = posix_spawn(&child, argv[1], &actions, &attr, argv + 1, environ);
  posix_spawn_file_actions_destroy(&actions);
  posix_spawnattr_destroy(&attr);
  if (error) { dprintf(3, "failed\n"); return 68; }
  if (dprintf(3, "child:%d\n", child) < 0) return 69;
  int status = 0;
  pid_t waited;
  do { waited = waitpid(child, &status, 0); } while (waited < 0 && errno == EINTR);
  if (waited != child) return 70;
  if (WIFEXITED(status)) dprintf(3, "exit:%d\n", WEXITSTATUS(status));
  else if (WIFSIGNALED(status)) dprintf(3, "signal:%d\n", WTERMSIG(status));
  else return 71;
  return 0;
}
