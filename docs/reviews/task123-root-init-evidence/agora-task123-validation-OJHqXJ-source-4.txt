// Real read-only native inspection in approved disposable directories; no mocks.
import { execFileSync, spawnSync } from 'node:child_process';
import { createHash } from 'node:crypto';
import {
  existsSync,
  lstatSync,
  mkdirSync,
  mkdtempSync,
  readdirSync,
  readFileSync,
  renameSync,
  rmSync,
  statfsSync,
  unlinkSync,
  writeFileSync,
} from 'node:fs';
import { basename, join, resolve } from 'node:path';
import { expect, it } from 'vitest';
import { initializeLocalRoot } from '../src/local-root-initialization';
import { inspectSelectedLocalRoot } from '../src/local-root-inspection';

for (const scenario of [
  'normal',
  'existing',
  'revoke',
  'move-before',
  'move-after',
  'replay',
  'missing-receipt',
])
  it(`initializes a registered staging directory without adopting existing data: ${scenario}`, async () => {
    const base = mkdtempSync('/private/tmp/agora-task123-validation-'),
      identity = lstatSync(base);
    const directory = join(base, 'project'),
      helper = join(base, 'inspect-root');
    mkdirSync(directory);
    writeFileSync(join(directory, 'sentinel'), 'fixed source');
    writeFileSync(join(base, 'external-secret'), 'fixed external secret');
    const source = resolve('packages/runtime/sandbox/native/local-root-inspection.c');
    const folder = resolve('docs/reviews/task123-root-init-evidence');
    mkdirSync(folder, { recursive: true });
    const evidence: Record<string, unknown> = {
      base,
      identity: { dev: identity.dev, ino: identity.ino },
      startedAt: new Date().toISOString(),
      scenario,
    };
    let error: unknown;
    try {
      const initSource = resolve('packages/runtime/sandbox/native/local-root-initialization.c');
      const initializer = join(base, 'initialize-root');
      for (const [sourcePath, output] of [
        [source, helper],
        [initSource, initializer],
      ]) {
        execFileSync('/usr/bin/clang', [
          '-std=c11',
          '-Wall',
          '-Wextra',
          '-Werror',
          '-mmacosx-version-min=15.0',
          sourcePath as string,
          '-o',
          output as string,
        ]);
      }
      const inspection = inspectSelectedLocalRoot(directory, helper);
      const journalRoot = join(base, 'journal');
      mkdirSync(journalRoot, { mode: 0o700 });
      const staging = join(directory, '.agora-operations');
      const moved = join(base, 'moved');
      if (scenario === 'existing') {
        mkdirSync(staging, { mode: 0o700 });
        writeFileSync(join(staging, 'user'), 'preserve');
      }
      const request = {
        actionId: 'fixed-init',
        inspection,
        helper: initializer,
        journalRoot,
        authorityHash: 'a'.repeat(64),
        authorize: async (checkpoint: string) => {
          if (checkpoint === 'before_initialize' && scenario === 'revoke') return false;
          if (checkpoint === 'before_initialize' && scenario === 'move-before')
            renameSync(directory, moved);
          if (checkpoint === 'after_initialize' && scenario === 'move-after')
            renameSync(directory, moved);
          return true;
        },
      };
      const result = await initializeLocalRoot(request);
      evidence.result = result;
      if (['normal', 'replay', 'missing-receipt'].includes(scenario)) {
        expect(result.stage).toBe('applied');
        expect(result.created).toBe(true);
        expect(result.stagingIdentity).toBe(`${lstatSync(staging).dev}:${lstatSync(staging).ino}`);
        expect(lstatSync(staging).mode & 0o777).toBe(0o700);
        if (scenario === 'replay') {
          expect(await initializeLocalRoot(request)).toEqual(result);
          await expect(
            initializeLocalRoot({ ...request, authorityHash: 'b'.repeat(64) }),
          ).rejects.toThrow('operation_conflict');
        }
        if (scenario === 'missing-receipt') {
          evidence.originalResult = readFileSync(join(result.journalPath, 'result.json'), 'utf8');
          unlinkSync(join(result.journalPath, 'result.json'));
          await expect(initializeLocalRoot(request)).rejects.toThrow('recovery_required');
        }
      } else if (scenario === 'existing') {
        expect(result.stage).toBe('conflict');
        expect(result.created).toBe(false);
        expect(readFileSync(join(staging, 'user'), 'utf8')).toBe('preserve');
      } else {
        expect(result.stage).toBe('recoveryRequired');
        expect(result.created).toBe(scenario === 'move-after');
        expect(existsSync(join(existsSync(moved) ? moved : directory, '.agora-operations'))).toBe(
          scenario === 'move-after',
        );
      }
      expect(readFileSync(join(base, 'external-secret'), 'utf8')).toBe('fixed external secret');
      expect(readFileSync(join(existsSync(moved) ? moved : directory, 'sentinel'), 'utf8')).toBe(
        'fixed source',
      );
      evidence.journal = Object.fromEntries(
        readdirSync(result.journalPath).map((name) => [
          name,
          readFileSync(join(result.journalPath, name), 'utf8'),
        ]),
      );
      evidence.binaryHash = createHash('sha256').update(readFileSync(initializer)).digest('hex');
      evidence.passed = true;
    } catch (e) {
      error = e;
      evidence.error = e instanceof Error ? { message: e.message, cause: e.cause } : String(e);
    }
    evidence.sources = Object.fromEntries(
      [
        'packages/runtime/sandbox/native/local-root-inspection.c',
        'packages/runtime/sandbox/native/local-root-initialization.c',
        'packages/runtime/sandbox/src/local-root-inspection.ts',
        'packages/runtime/sandbox/src/local-root-initialization.ts',
        'packages/runtime/sandbox/test/local-root-initialization.test.ts',
      ].map((path, index) => {
        const data = existsSync(path) ? readFileSync(path) : Buffer.from('not implemented');
        const snapshot = join(folder, `${basename(base)}-source-${index}.txt`);
        writeFileSync(snapshot, data);
        return [path, { sha256: createHash('sha256').update(data).digest('hex'), snapshot }];
      }),
    );
    const output = join(folder, `${basename(base)}.json`);
    writeFileSync(output, JSON.stringify(evidence, null, 2));
    const current = lstatSync(base),
      handles = spawnSync('/usr/sbin/lsof', ['-nP', '+D', base], { encoding: 'utf8' }),
      mounts = spawnSync('/sbin/mount', [], { encoding: 'utf8' });
    if (
      current.dev !== identity.dev ||
      current.ino !== identity.ino ||
      handles.status !== 1 ||
      handles.stdout ||
      handles.stderr ||
      mounts.status !== 0 ||
      mounts.stdout.includes(base)
    )
      throw new Error('root_fixture_cleanup_unproven');
    const before = statfsSync(base);
    rmSync(base, { recursive: true });
    const after = statfsSync('/private/tmp');
    evidence.cleanup = {
      deleted: true,
      noOpenHandles: true,
      noMounts: true,
      availableBytesDelta: after.bavail * after.bsize - before.bavail * before.bsize,
    };
    writeFileSync(output, JSON.stringify(evidence, null, 2));
    if (error) throw error;
  }, 20000);
