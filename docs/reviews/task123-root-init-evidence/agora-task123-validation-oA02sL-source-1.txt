/* Trusted exclusive staging initialization. No project code or recursive access. */
#define _DARWIN_C_SOURCE
#include <errno.h>
#include <fcntl.h>
#include <inttypes.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/mount.h>
#include <sys/stat.h>
#include <unistd.h>
#define DEPTH 128
static int fds[DEPTH], count, created, staging=-1;
static struct stat ids[DEPTH], staging_id;
static char names[DEPTH][256];
static void finish(const char *stage,const char *reason,int code) {
  printf("{\"event\":\"result\",\"stage\":\"%s\",\"reason\":\"%s\",\"created\":%s,\"stagingIdentity\":",stage,reason,created?"true":"false");
  if(staging>=0) printf("\"%ju:%ju\"",(uintmax_t)staging_id.st_dev,(uintmax_t)staging_id.st_ino); else printf("null");
  puts("}");fflush(stdout);exit(code);
}
static void fail(const char *reason) {finish("recoveryRequired",reason,1);}
static void verify(void) {
  for(int i=0;i<count;i++) {
    struct stat s;
    if((i?fstatat(fds[i-1],names[i],&s,AT_SYMLINK_NOFOLLOW):lstat("/",&s)) ||
       !S_ISDIR(s.st_mode) || s.st_dev!=ids[i].st_dev || s.st_ino!=ids[i].st_ino ||
       s.st_mode!=ids[i].st_mode || s.st_uid!=ids[i].st_uid || (s.st_flags & SF_DATALESS)) fail("root_identity_changed");
  }
  if(staging>=0) {
    struct stat s;
    if(fstatat(fds[count-1],".agora-operations",&s,AT_SYMLINK_NOFOLLOW) ||
       !S_ISDIR(s.st_mode) || s.st_dev!=staging_id.st_dev || s.st_ino!=staging_id.st_ino ||
       s.st_uid!=getuid() || (s.st_mode&0777)!=0700) fail("staging_identity_changed");
  }
}
static void checkpoint(const char *name) {
  printf("{\"event\":\"checkpoint\",\"name\":\"%s\"}\n",name);fflush(stdout);
  char c;
  if(read(0,&c,1)!=1 || c!='x') fail("authorization_closed");
  verify();
}
static void append(int fd,const char *name,char **expected) {
  if(fd<0 || count>=DEPTH || strlen(name)>255 || fstat(fd,&ids[count]) || !S_ISDIR(ids[count].st_mode) ||
     (ids[count].st_flags&SF_DATALESS)) fail("root_identity_changed");
  char *next=strchr(*expected,',');if(next) *next=0;
  char identity[80];snprintf(identity,sizeof(identity),"%ju:%ju",(uintmax_t)ids[count].st_dev,(uintmax_t)ids[count].st_ino);
  if(strcmp(identity,*expected)) fail("root_identity_changed");
  *expected=next?next+1:*expected+strlen(*expected);
  fds[count]=fd;strcpy(names[count],name);count++;
}
int main(int argc,char **argv) {
  if(argc!=3 || argv[1][0]!='/' || !argv[1][1] || strlen(argv[1])>=4096 || strlen(argv[2])>=16384) return 64;
  char path[4096],expected[16384];strcpy(path,argv[1]+1);strcpy(expected,argv[2]);
  char *identity=expected;
  append(open("/",O_RDONLY|O_DIRECTORY|O_NOFOLLOW|O_CLOEXEC),"",&identity);
  char *name=path;
  for(;;) {
    char *next=strchr(name,'/');if(next)*next=0;
    if(!*name || !strcmp(name,".") || !strcmp(name,"..")) fail("invalid_request");
    append(openat(fds[count-1],name,O_RDONLY|O_DIRECTORY|O_NOFOLLOW|O_CLOEXEC),name,&identity);
    if(!next)break;
    name=next+1;
  }
  if(*identity)fail("invalid_request");
  struct statfs fs;
  if(fstatfs(fds[count-1],&fs) || strcmp(fs.f_fstypename,"apfs") || !(fs.f_flags&MNT_LOCAL)) fail("unsupported_workspace");
  checkpoint("before_initialize");
  if(mkdirat(fds[count-1],".agora-operations",0700)) {
    if(errno==EEXIST)finish("conflict","staging_exists",1);
    fail("initialization_failed");
  }
  created=1;
  verify();
  staging=openat(fds[count-1],".agora-operations",O_RDONLY|O_DIRECTORY|O_NOFOLLOW|O_CLOEXEC);
  if(staging<0 || fstat(staging,&staging_id)) {staging=-1;fail("staging_identity_changed");}
  verify();
  if(fsync(staging) || fsync(fds[count-1]))fail("flush_failed");
  checkpoint("after_initialize");
  finish("applied","none",0);
}
