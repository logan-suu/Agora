/** Internal observation and registered-cohort primitives. Durable membership,
 * continuous discovery, root binding and admission remain the supervisor's responsibility.
 * Success here cannot issue a full cleanup or resource-reuse receipt. */
import { execFileSync } from 'node:child_process';
import { performance } from 'node:perf_hooks';
import { setTimeout as delay } from 'node:timers/promises';

export type LocalProcessIdentity = readonly number[];
type ProcessState = 'alive' | 'exited' | 'identityMismatch' | 'unknown';
type Observation = { state: ProcessState; signalSent: boolean; errno: number };
export type RegisteredStopReceipt = {
  assurance: 'registered-only';
  registeredState: 'stopped' | 'needsAttention';
  signals: ('TERM' | 'KILL')[];
  durationMs: number;
  states: ProcessState[];
};

function invoke(helper: string, args: string[], timeout: number): unknown {
  if (process.platform !== 'darwin') throw new Error('sandbox_unavailable');
  return JSON.parse(
    execFileSync(helper, args, {
      encoding: 'utf8',
      timeout: Math.max(1, Math.min(250, Math.floor(timeout))),
      maxBuffer: 32768,
      env: { NODE_ENV: 'production' },
      stdio: ['ignore', 'pipe', 'pipe'],
    }),
  );
}

function validIdentity(value: unknown): value is LocalProcessIdentity {
  return (
    Array.isArray(value) &&
    value.length === 8 &&
    value.every((word) => Number.isInteger(word) && word >= 0 && word <= 0xffffffff) &&
    value[5] > 1 &&
    value[5] <= 0x7fffffff &&
    value[1] === process.getuid?.()
  );
}

/** Only the trusted launcher may call this after proving child membership. */
export function captureLocalProcess(helper: string, pid: number): LocalProcessIdentity {
  if (!Number.isSafeInteger(pid) || pid <= 1) throw new Error('invalid_process_identity');
  const result = invoke(helper, ['capture', String(pid)], 250) as { token?: unknown };
  if (!validIdentity(result?.token) || result.token[5] !== pid)
    throw new Error('invalid_process_identity');
  return Object.freeze([...result.token]);
}

export function inspectLocalProcess(
  helper: string,
  identity: LocalProcessIdentity,
  signal?: 'TERM' | 'KILL',
  timeout = 250,
): Observation {
  if (!validIdentity(identity)) throw new Error('invalid_process_identity');
  try {
    const args = [signal ? 'signal' : 'inspect', ...identity.map(String)];
    if (signal) args.push(signal === 'TERM' ? '15' : '9');
    const result = invoke(helper, args, timeout) as Observation;
    if (
      !result ||
      !['alive', 'exited', 'identityMismatch', 'unknown'].includes(result.state) ||
      typeof result.signalSent !== 'boolean' ||
      !Number.isInteger(result.errno)
    )
      throw new Error('invalid_process_observation');
    return result;
  } catch {
    return { state: 'unknown', signalSent: false, errno: 0 };
  }
}

export type LocalProcessDiscovery = {
  assurance: 'bounded-observation';
  observationState: 'observed' | 'needsAttention';
  identities: LocalProcessIdentity[];
  relations: { parent: LocalProcessIdentity; child: LocalProcessIdentity }[];
  durationMs: number;
};

/** Seeds must already belong to this command. A finite scan is not a cessation
 * proof; the caller must persist new membership before using it for cleanup. */
export function discoverLocalProcessCohort(
  helper: string,
  seeds: readonly LocalProcessIdentity[],
  outerDeadline = Number.POSITIVE_INFINITY,
): LocalProcessDiscovery {
  if (
    !seeds.length ||
    seeds.length > 256 ||
    seeds.some((entry) => !validIdentity(entry)) ||
    new Set(seeds.map((entry) => entry.join(':'))).size !== seeds.length
  )
    throw new Error('invalid_process_cohort');
  const started = performance.now();
  const deadline = Math.min(started + 1000, outerDeadline);
  const identities = seeds.map((entry) => Object.freeze([...entry]));
  const seen = new Set(identities.map((entry) => entry.join(':')));
  const relations: LocalProcessDiscovery['relations'] = [];
  let uncertain = false;
  for (let index = 0; index < identities.length; index++) {
    const parent = identities[index] as LocalProcessIdentity;
    const remaining = deadline - performance.now();
    if (remaining < 250) {
      uncertain = true;
      break;
    }
    try {
      const result = invoke(helper, ['children', ...parent.map(String)], remaining) as {
        state?: unknown;
        children?: unknown;
      };
      if (
        !result ||
        !['observed', 'exited', 'identityMismatch', 'unknown'].includes(String(result.state)) ||
        !Array.isArray(result.children) ||
        result.children.length > 256 ||
        result.children.some((child) => !validIdentity(child)) ||
        (result.state !== 'observed' && result.state !== 'unknown' && result.children.length > 0)
      )
        throw new Error('invalid_discovery_observation');
      if (result.state !== 'observed') uncertain = true;
      for (const child of result.children as LocalProcessIdentity[]) {
        const key = child.join(':');
        if (seen.has(key)) continue;
        if (identities.length === 256) {
          uncertain = true;
          break;
        }
        const identity = Object.freeze([...child]);
        identities.push(identity);
        seen.add(key);
        relations.push({ parent, child: identity });
      }
    } catch {
      uncertain = true;
    }
  }
  return {
    assurance: 'bounded-observation',
    observationState: uncertain ? 'needsAttention' : 'observed',
    identities,
    relations,
    durationMs: performance.now() - started,
  };
}

export async function stopRegisteredLocalProcesses(
  helper: string,
  identities: readonly LocalProcessIdentity[],
  outerDeadline = Number.POSITIVE_INFINITY,
): Promise<RegisteredStopReceipt> {
  if (identities.length > 256 || identities.some((entry) => !validIdentity(entry)))
    throw new Error('invalid_process_cohort');
  const cohort = identities.map((entry) => [...entry]);
  if (new Set(cohort.map((entry) => entry.join(':'))).size !== cohort.length)
    throw new Error('duplicate_process_identity');
  const started = performance.now();
  const deadline = Math.min(started + 5000, outerDeadline);
  const states: ProcessState[] = cohort.map(() => 'alive');
  const signals: ('TERM' | 'KILL')[] = [];
  let uncertain = false;
  function visit(signal?: 'TERM' | 'KILL', until = deadline) {
    for (const [index, identity] of cohort.entries()) {
      if (states[index] !== 'alive') continue;
      const remaining = Math.min(deadline, until) - performance.now();
      // A partial launch window can kill an otherwise healthy observer just
      // before TERM expires and incorrectly suppress the KILL phase. Reserve
      // its full bound; let the phase advance when that budget is unavailable.
      if (remaining < 250) {
        if (performance.now() >= deadline) uncertain = true;
        return;
      }
      const observation = inspectLocalProcess(helper, identity, signal, remaining);
      states[index] = observation.state;
      if (observation.state === 'unknown' || observation.state === 'identityMismatch')
        uncertain = true;
      if (signal && observation.signalSent && !signals.includes(signal)) signals.push(signal);
    }
  }
  async function settle(until: number) {
    while (states.includes('alive') && performance.now() < until) {
      visit(undefined, until);
      if (states.includes('alive'))
        await delay(Math.max(0, Math.min(25, until - performance.now())));
    }
  }
  const termDeadline = Math.min(deadline, started + 1000);
  visit('TERM', termDeadline);
  await settle(termDeadline);
  const killDeadline = Math.min(deadline, performance.now() + 2000);
  visit('KILL', killDeadline);
  await settle(killDeadline);
  return {
    assurance: 'registered-only',
    registeredState:
      !uncertain && states.every((state) => state === 'exited') ? 'stopped' : 'needsAttention',
    signals,
    durationMs: performance.now() - started,
    states,
  };
}
