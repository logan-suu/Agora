// Real kernel process relations and Seatbelt payloads; no test doubles.
import { expect, it } from 'vitest';
import { discoverLocalProcessCohort } from '../../../packages/runtime/sandbox/src/local-command-stop';
import { probeLocalCommandDiscovery } from './local-command-discovery-fixture';

it('discovers a real child and grandchild without enrolling an unrelated sibling', async () => {
  const result = await probeLocalCommandDiscovery('tree');
  expect(result.discovery.observationState).toBe('observed');
  expect(result.discovery.assurance).toBe('bounded-observation');
  expect(result.discovery.identities).toHaveLength(3);
  expect(result.discovery.relations).toHaveLength(2);
  expect(result.discovery.relations[0]?.parent).toEqual(result.discovery.identities[0]);
  expect(result.discovery.relations[1]?.parent).toEqual(result.discovery.relations[0]?.child);
  expect(result.stop.registeredState).toBe('stopped');
  expect(result.afterStop.observationState).toBe('needsAttention');
  expect(result.afterStop.relations).toEqual([]);
  expect(result.siblingAlive).toBe(true);
  expect(result.persistedBirths).toEqual(result.discovery.identities);
  expect(result.blocked).toBe(true);
}, 20_000);

it('rejects empty, invalid and duplicate seed identities before inspecting any process', () => {
  const token = [0, process.getuid?.() ?? 0, 0, 0, 0, 2, 0, 1];
  expect(() => discoverLocalProcessCohort('/absent', [])).toThrow('invalid_process_cohort');
  expect(() => discoverLocalProcessCohort('/absent', [[1]])).toThrow('invalid_process_cohort');
  expect(() => discoverLocalProcessCohort('/absent', [token, token])).toThrow(
    'invalid_process_cohort',
  );
});

it('retains a discovered grandchild after its parent exits and it is reparented', async () => {
  const result = await probeLocalCommandDiscovery('reparent');
  expect(result.discovery.observationState).toBe('observed');
  expect(result.discovery.identities).toHaveLength(3);
  expect(result.reparented).toBe(true);
  expect(result.stop.registeredState).toBe('stopped');
  expect(result.siblingAlive).toBe(true);
  expect(result.blocked).toBe(true);
}, 20_000);

it('rejects a wrong parent birth and leaves the real tree alive', async () => {
  const result = await probeLocalCommandDiscovery('wrong-parent');
  expect(result.rejected?.observationState).toBe('needsAttention');
  expect(result.rejected?.relations).toEqual([]);
  expect(result.treeAliveAfterRejection).toBe(true);
  expect(result.stop.registeredState).toBe('stopped');
}, 20_000);

it('exposes helper failure as incomplete discovery without signaling the root', async () => {
  const result = await probeLocalCommandDiscovery('missing-helper');
  expect(result.rejected?.observationState).toBe('needsAttention');
  expect(result.rejected?.relations).toEqual([]);
  expect(result.treeAliveAfterRejection).toBe(true);
  expect(result.stop.registeredState).toBe('stopped');
}, 20_000);
