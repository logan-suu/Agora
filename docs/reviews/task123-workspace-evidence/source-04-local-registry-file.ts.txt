/** Internal durable file adapter. A trusted control service must provide the
 * complete record codec and authorization/transition checks. This adapter alone
 * cannot grant access, close a TaskState operation or launch a workspace. */
import { createHash, randomUUID } from 'node:crypto';
import { constants } from 'node:fs';
import {
  type FileHandle,
  lstat,
  mkdir,
  open,
  readdir,
  realpath,
  rename,
  unlink,
} from 'node:fs/promises';
import { dirname, isAbsolute, join } from 'node:path';
import { assertWorkspaceRefsTransition, isWorkspaceRefsV1 } from '@agora/core-domain';

export interface LocalRegistryOwner {
  readonly root: string;
  assertHeld(): Promise<void>;
}
export interface LocalRegistrySnapshot {
  schemaVersion: 'local-workspaces-v1';
  revision: number;
  roots: unknown[];
  grants: unknown[];
  workspaces: unknown[];
  claims: unknown[];
  operations: unknown[];
}

type Identity = { path: string; dev: number; ino: number; uid: number; mode: number };
const maxBytes = 16 * 1024 * 1024;
const digest = (bytes: string | Buffer) => createHash('sha256').update(bytes).digest('hex');
const keys = ['schemaVersion', 'revision', 'roots', 'grants', 'workspaces', 'claims', 'operations'];
const arrays = ['roots', 'grants', 'workspaces', 'claims', 'operations'] as const;
const safeRevision = (value: unknown): value is number =>
  typeof value === 'number' && Number.isSafeInteger(value) && value >= 0;

async function directory(path: string): Promise<Identity> {
  const stat = await lstat(path);
  if (!stat.isDirectory() || stat.isSymbolicLink()) throw new Error('registry_root_changed');
  return { path, dev: stat.dev, ino: stat.ino, uid: stat.uid, mode: stat.mode };
}
async function syncDirectory(path: string) {
  const fd = await open(path, constants.O_RDONLY | constants.O_DIRECTORY | constants.O_NOFOLLOW);
  try {
    await fd.sync();
  } finally {
    await fd.close();
  }
}

async function readBounded(fd: FileHandle) {
  const chunks: Buffer[] = [];
  const buffer = Buffer.alloc(64 * 1024);
  let length = 0;
  for (;;) {
    const { bytesRead } = await fd.read(buffer);
    if (bytesRead === 0) return Buffer.concat(chunks, length).toString('utf8');
    length += bytesRead;
    if (length > maxBytes) throw new Error('invalid_local_registry');
    chunks.push(Buffer.from(buffer.subarray(0, bytesRead)));
  }
}

export class LocalRegistryFile {
  private readonly file: string;
  private readonly lock: string;
  private readonly chain: Identity[] = [];

  private constructor(
    private readonly owner: LocalRegistryOwner,
    private readonly validateRecords: (snapshot: unknown) => unknown,
    private readonly root: string,
  ) {
    this.file = join(root, 'registry.json');
    this.lock = join(root, 'registry.lock');
  }

  /** Creation is explicit. An existing incomplete directory is never reset. */
  static async open(
    owner: LocalRegistryOwner,
    validateRecords: (snapshot: unknown) => unknown,
    initialize = false,
  ) {
    if (typeof validateRecords !== 'function') throw new Error('registry_codec_required');
    await owner.assertHeld();
    if (
      !isAbsolute(owner.root) ||
      owner.root === '/' ||
      (await realpath(owner.root)) !== owner.root
    )
      throw new Error('registry_root_changed');
    const parent = await directory(owner.root);
    if (parent.uid !== process.getuid?.() || (parent.mode & 0o777) !== 0o700)
      throw new Error('registry_root_changed');
    const root = join(owner.root, 'local-workspaces');
    let created = false;
    if (initialize) {
      try {
        await mkdir(root, { mode: 0o700 });
        created = true;
        await syncDirectory(owner.root);
      } catch (error) {
        if ((error as NodeJS.ErrnoException).code !== 'EEXIST') throw error;
      }
    }
    const store = new LocalRegistryFile(owner, validateRecords, root);
    for (let path = root; ; path = dirname(path)) {
      store.chain.push(await directory(path));
      if (path === '/') break;
    }
    const captured = store.chain[1];
    if (captured?.dev !== parent.dev || captured.ino !== parent.ino)
      throw new Error('registry_root_changed');
    await store.assertRoot();
    if (created) {
      const initial = store.validate({
        schemaVersion: 'local-workspaces-v1',
        revision: 0,
        roots: [],
        grants: [],
        workspaces: [],
        claims: [],
        operations: [],
      });
      await store.write(undefined, initial);
    } else await store.load();
    return store;
  }

  private validate(value: unknown): LocalRegistrySnapshot {
    try {
      if (!value || !isWorkspaceRefsV1((value as LocalRegistrySnapshot).workspaces))
        throw new Error();
      const bytes = JSON.stringify(value);
      if (typeof bytes !== 'string' || Buffer.byteLength(bytes) > maxBytes) throw new Error();
      const cloned = JSON.parse(bytes) as LocalRegistrySnapshot;
      if (
        !cloned ||
        Array.isArray(cloned) ||
        Object.keys(cloned).length !== keys.length ||
        !keys.every((key) => Object.hasOwn(cloned, key)) ||
        cloned.schemaVersion !== 'local-workspaces-v1' ||
        !safeRevision(cloned.revision) ||
        !isWorkspaceRefsV1(cloned.workspaces) ||
        arrays.some((key) => !Array.isArray(cloned[key]) || cloned[key].length > 4096)
      )
        throw new Error();
      // Record parsing is mandatory and must reject rather than silently migrate/normalize.
      if (JSON.stringify(this.validateRecords(structuredClone(cloned))) !== bytes)
        throw new Error();
      return cloned;
    } catch {
      throw new Error('invalid_local_registry');
    }
  }

  private async assertRoot() {
    await this.owner.assertHeld();
    try {
      for (const prior of this.chain) {
        const current = await directory(prior.path);
        if (
          current.dev !== prior.dev ||
          current.ino !== prior.ino ||
          current.uid !== prior.uid ||
          current.mode !== prior.mode
        )
          throw new Error();
      }
      const root = this.chain[0];
      if (!root || root.uid !== process.getuid?.() || (root.mode & 0o777) !== 0o700)
        throw new Error();
    } catch {
      throw new Error('registry_root_changed');
    }
    await this.owner.assertHeld();
  }

  private async assertNoPending(locked: boolean) {
    const names = await readdir(this.root);
    if (
      names.some((name) => name.startsWith('registry.pending-')) ||
      (!locked && names.includes('registry.lock'))
    )
      throw new Error('registry_recovery_required');
  }

  private async read(locked = false): Promise<LocalRegistrySnapshot> {
    await this.assertRoot();
    await this.assertNoPending(locked);
    let fd: FileHandle | undefined;
    try {
      fd = await open(this.file, constants.O_RDONLY | constants.O_NOFOLLOW);
      const stat = await fd.stat();
      if (
        !stat.isFile() ||
        stat.nlink !== 1 ||
        stat.uid !== process.getuid?.() ||
        (stat.mode & 0o777) !== 0o600 ||
        stat.size > maxBytes
      )
        throw new Error();
      const wrapper = JSON.parse(await readBounded(fd));
      if (
        !wrapper ||
        Object.keys(wrapper).length !== 2 ||
        !Object.hasOwn(wrapper, 'snapshot') ||
        !Object.hasOwn(wrapper, 'sha256') ||
        wrapper.sha256 !== digest(JSON.stringify(wrapper.snapshot))
      )
        throw new Error();
      const result = this.validate(wrapper.snapshot);
      const after = await fd.stat();
      const current = await lstat(this.file);
      if (
        after.dev !== stat.dev ||
        after.ino !== stat.ino ||
        after.size !== stat.size ||
        after.mtimeMs !== stat.mtimeMs ||
        after.ctimeMs !== stat.ctimeMs ||
        current.isSymbolicLink() ||
        current.dev !== stat.dev ||
        current.ino !== stat.ino
      )
        throw new Error();
      await this.assertRoot();
      await this.assertNoPending(locked);
      return result;
    } catch (error) {
      if (
        error instanceof Error &&
        ['registry_recovery_required', 'registry_root_changed', 'state_owner_changed'].includes(
          error.message,
        )
      )
        throw error;
      throw new Error('invalid_local_registry');
    } finally {
      await fd?.close();
    }
  }

  async load() {
    return this.read();
  }

  async compareAndSwap(expectedRevision: number, next: LocalRegistrySnapshot) {
    if (!safeRevision(expectedRevision) || !safeRevision(expectedRevision + 1))
      throw new Error('registry_revision_exhausted');
    // Snapshot before the first await; later caller mutation cannot alter publication.
    const snapshot = this.validate(next);
    if (snapshot.revision !== expectedRevision + 1) throw new Error('registry_revision_conflict');
    await this.write(expectedRevision, snapshot);
    return structuredClone(snapshot);
  }

  private async write(expectedRevision: number | undefined, snapshot: LocalRegistrySnapshot) {
    await this.assertRoot();
    await this.assertNoPending(false);
    let lock: FileHandle;
    try {
      lock = await open(this.lock, 'wx', 0o600);
    } catch {
      throw new Error('registry_recovery_required');
    }
    const lockIdentity = await lock.stat().catch(async () => {
      await lock.close();
      throw new Error('registry_recovery_required');
    });
    let removable = false;
    let publishing = false;
    try {
      await lock.writeFile(JSON.stringify({ version: 1, operationId: randomUUID() }));
      await lock.sync();
      await syncDirectory(this.root);
      await this.assertRoot();
      if (expectedRevision === undefined) {
        if ((await readdir(this.root)).some((name) => name !== 'registry.lock'))
          throw new Error('registry_already_exists');
      } else {
        const current = await this.read(true);
        if (current.revision !== expectedRevision) throw new Error('registry_revision_conflict');
        assertWorkspaceRefsTransition(current.workspaces, snapshot.workspaces);
      }
      const bytes = JSON.stringify({ snapshot, sha256: digest(JSON.stringify(snapshot)) });
      if (Buffer.byteLength(bytes) > maxBytes) throw new Error('invalid_local_registry');
      const pending = join(this.root, `registry.pending-${randomUUID()}`);
      publishing = true;
      const fd = await open(pending, 'wx', 0o600);
      try {
        await fd.writeFile(bytes);
        await fd.sync();
      } finally {
        await fd.close();
      }
      await this.assertRoot();
      const activeLock = await lstat(this.lock);
      if (
        !activeLock.isFile() ||
        activeLock.nlink !== 1 ||
        activeLock.uid !== process.getuid?.() ||
        (activeLock.mode & 0o777) !== 0o600 ||
        activeLock.isSymbolicLink() ||
        activeLock.dev !== lockIdentity.dev ||
        activeLock.ino !== lockIdentity.ino
      )
        throw new Error('registry_recovery_required');
      await rename(pending, this.file);
      await syncDirectory(this.root);
      await this.assertRoot();
      removable = true;
    } catch (error) {
      // Before publication, a rejected CAS has no partial snapshot to recover.
      if (!publishing) removable = true;
      throw error;
    } finally {
      await lock.close();
      if (removable) {
        await this.assertRoot();
        await this.removeLock(lockIdentity.dev, lockIdentity.ino);
      }
    }
  }

  private async removeLock(dev: number, ino: number) {
    const active = await lstat(this.lock);
    if (
      !active.isFile() ||
      active.nlink !== 1 ||
      active.uid !== process.getuid?.() ||
      (active.mode & 0o777) !== 0o600 ||
      active.isSymbolicLink() ||
      active.dev !== dev ||
      active.ino !== ino
    )
      throw new Error('registry_recovery_required');
    await unlink(this.lock);
    await syncDirectory(this.root);
  }
}
