/** Internal control-plane initialization. Grant creation remains the responsibility
 * of the authenticated Leader controller; this service cannot manufacture grants. */
import { createHash } from 'node:crypto';
import { closeSync, constants, fstatSync, lstatSync, openSync, readFileSync } from 'node:fs';
import { mkdir, open, realpath } from 'node:fs/promises';
import { join } from 'node:path';
import { parseWorkspaceControl } from '@agora/core-domain';
import type { LocalBindingCoordinator } from './local-binding-coordinator';
import { LocalRegistryFile, type LocalRegistryOwner } from './local-registry-file';
import {
  assertLocalRegistryTransition,
  isLocalBindingOperation,
  type LocalGrantRecord,
  type LocalRootInitializationOperation,
  type LocalRootRecord,
  localRecordHash,
  parseLocalRegistry,
} from './local-registry-records';
import {
  initializeLocalRoot,
  type LocalRootInitializationReceipt,
} from './local-root-initialization';
import { inspectSelectedLocalRoot } from './local-root-inspection';

type Request = {
  projectId: string;
  taskId: string;
  actionId: string;
  rootId: string;
  grantId: string;
  expectedRevision: number;
};
type Helpers = { inspector: string; initializer: string };
const digest = (data: Buffer | string) => createHash('sha256').update(data).digest('hex');
const id = (s: unknown) => typeof s === 'string' && /^[A-Za-z0-9][A-Za-z0-9._:-]{0,127}$/.test(s);
export class LocalRootCoordinator {
  private tail: Promise<unknown> = Promise.resolve();
  private constructor(
    private readonly registry: LocalRegistryFile,
    private readonly control: LocalBindingCoordinator,
    private readonly journalRoot: string,
    private readonly helpers: Helpers,
  ) {}
  static async open(owner: LocalRegistryOwner, control: LocalBindingCoordinator, helpers: Helpers) {
    const registry = await LocalRegistryFile.open(
      owner,
      parseLocalRegistry,
      false,
      assertLocalRegistryTransition,
    );
    const journalRoot = join(owner.root, 'local-workspaces', 'initializations');
    try {
      await mkdir(journalRoot, { mode: 0o700 });
      const fd = await open(
        join(owner.root, 'local-workspaces'),
        constants.O_RDONLY | constants.O_DIRECTORY | constants.O_NOFOLLOW,
      );
      try {
        await fd.sync();
      } finally {
        await fd.close();
      }
    } catch (error) {
      if ((error as NodeJS.ErrnoException).code !== 'EEXIST') throw error;
    }
    const stat = lstatSync(journalRoot);
    if (
      !stat.isDirectory() ||
      stat.isSymbolicLink() ||
      stat.uid !== process.getuid?.() ||
      (stat.mode & 0o777) !== 0o700 ||
      (await realpath(journalRoot)) !== journalRoot
    )
      throw Error('untrusted_runtime_path');
    await owner.assertHeld();
    return new LocalRootCoordinator(registry, control, journalRoot, { ...helpers });
  }
  initialize(input: Request): Promise<LocalRootInitializationOperation> {
    localRecordHash(input);
    if (
      Object.keys(input).sort().join(',') !==
        'actionId,expectedRevision,grantId,projectId,rootId,taskId' ||
      !['projectId', 'taskId', 'actionId', 'rootId', 'grantId'].every((k) =>
        id(input[k as keyof Request]),
      ) ||
      !Number.isSafeInteger(input.expectedRevision) ||
      input.expectedRevision < 0
    )
      return Promise.reject(Error('workspace_initialization_input_invalid'));
    const request = structuredClone(input);
    const next = this.tail.catch(() => undefined).then(() => this.run(request));
    this.tail = next.catch(() => undefined);
    return next;
  }
  private async source(
    request: Request,
    root: LocalRootRecord,
    grant: LocalGrantRecord,
    operation?: LocalRootInitializationOperation,
  ) {
    const state =
      operation?.stage === 'prepared'
        ? await this.control.assertInitializationSource(operation)
        : await this.control.assertClosed(request);
    const registry = parseLocalRegistry(await this.registry.load());
    const origin = registry.operations
      .filter(isLocalBindingOperation)
      .find((o) => o.actionId === grant.createdActionId);
    const message = state.messages.find((m) => m.msgId === grant.leaderMessageId);
    let command;
    try {
      command = message ? parseWorkspaceControl(message.display.trim()) : undefined;
    } catch {
      /* Invalid source is never authorization. */
    }
    if (
      !origin ||
      origin.stage !== 'committed' ||
      origin.projectId !== request.projectId ||
      origin.taskId !== request.taskId ||
      origin.sourceMessageId !== grant.leaderMessageId ||
      !origin.nextLocalExecution.rootIds.includes(root.rootId) ||
      !message ||
      message.channelId !== 'main' ||
      message.fromRole !== 'leader' ||
      message.payload.kind !== 'leader_intent' ||
      !command ||
      command.verb !== 'grant' ||
      command.projectId !== request.projectId ||
      command.taskId !== request.taskId ||
      command.actionId !== grant.createdActionId ||
      command.selectionRef !== root.selectionRef ||
      command.expectedRevision + 1 !== origin.preparedRevision ||
      localRecordHash(message.payload.intent) !== localRecordHash(command) ||
      localRecordHash(message.payload.action) !== localRecordHash({ status: 'applied' })
    )
      throw Error('workspace_grant_source_invalid');
    if (grant.status !== 'active' || !grant.actions.includes('edit'))
      throw Error('workspace_grant_closed');
    return { registry, sourceMessageId: message.msgId };
  }
  private nativeReceipt(path: string, expected: LocalRootInitializationReceipt) {
    const fd = openSync(path, constants.O_RDONLY | constants.O_NOFOLLOW);
    try {
      const stat = fstatSync(fd);
      if (
        !stat.isFile() ||
        stat.nlink !== 1 ||
        stat.uid !== process.getuid?.() ||
        (stat.mode & 0o777) !== 0o400 ||
        stat.size > 16384
      )
        throw Error('root_initialization_receipt_invalid');
      const bytes = readFileSync(fd);
      if (localRecordHash(JSON.parse(bytes.toString('utf8'))) !== localRecordHash(expected))
        throw Error('root_initialization_receipt_invalid');
      return {
        sha256: digest(bytes),
        stage: expected.stage,
        created: expected.created,
        stagingIdentity: expected.stagingIdentity,
        quiescent: expected.quiescent,
      };
    } finally {
      closeSync(fd);
    }
  }
  private async run(request: Request): Promise<LocalRootInitializationOperation> {
    let registry = parseLocalRegistry(await this.registry.load());
    const root = registry.roots.find(
      (r) => r.rootId === request.rootId && r.projectId === request.projectId,
    );
    const grant = registry.grants.find(
      (g) =>
        g.grantId === request.grantId &&
        g.rootId === request.rootId &&
        g.projectId === request.projectId,
    );
    if (!root || !grant) throw Error('workspace_initialization_scope_invalid');
    const existing = registry.operations.find((o) => o.actionId === request.actionId);
    if (existing && isLocalBindingOperation(existing)) throw Error('operation_conflict');
    const inputHash = localRecordHash({
      request,
      root: { ...root, staging: null },
      grant,
      helpers: {
        inspector: digest(readFileSync(this.helpers.inspector)),
        initializer: digest(readFileSync(this.helpers.initializer)),
      },
    });
    if (existing && existing.inputHash !== inputHash) throw Error('operation_conflict');
    const source = await this.source(request, root, grant, existing);
    if (source.registry.revision !== registry.revision) throw Error('registry_revision_conflict');
    const inspection = inspectSelectedLocalRoot(root.path, this.helpers.inspector);
    if (localRecordHash(inspection) !== root.inspectionHash) throw Error('root_identity_changed');
    let operation = existing;
    if (!operation) {
      if (root.staging !== null) throw Error('root_already_initialized');
      if (registry.revision !== request.expectedRevision) throw Error('registry_revision_conflict');
      const receiptId = `root-init:${digest(request.actionId)}`;
      operation = {
        kind: 'root-initialization',
        actionId: request.actionId,
        inputHash,
        receiptId,
        projectId: request.projectId,
        taskId: request.taskId,
        rootId: root.rootId,
        grantId: grant.grantId,
        grantRevision: grant.revision,
        sourceMessageId: source.sourceMessageId,
        preparedRevision: registry.revision + 1,
        stage: 'prepared',
        nativeReceipt: null,
      };
      await this.registry.compareAndSwap(registry.revision, {
        ...registry,
        revision: registry.revision + 1,
        operations: [...registry.operations, operation],
      });
    }
    const bound = operation;
    const nativeActionId = `init-${digest(request.actionId)}`;
    const authorityHash = localRecordHash({
      inputHash,
      grant,
      sourceMessageId: source.sourceMessageId,
    });
    const result = await initializeLocalRoot({
      actionId: nativeActionId,
      inspection,
      helper: this.helpers.initializer,
      journalRoot: this.journalRoot,
      authorityHash,
      authorize: async () => {
        try {
          const now = parseLocalRegistry(await this.registry.load());
          const current = now.operations.find((o) => o.actionId === bound.actionId);
          const currentGrant = now.grants.find((g) => g.grantId === grant.grantId);
          const currentRoot = now.roots.find((r) => r.rootId === root.rootId);
          if (
            !current ||
            isLocalBindingOperation(current) ||
            localRecordHash(current) !== localRecordHash(bound) ||
            !currentGrant ||
            localRecordHash(currentGrant) !== localRecordHash(grant) ||
            !currentRoot ||
            localRecordHash({ ...currentRoot, staging: null }) !==
              localRecordHash({ ...root, staging: null })
          )
            return false;
          await this.source(request, currentRoot, currentGrant, current);
          return true;
        } catch {
          return false;
        }
      },
    });
    const nativeReceipt = this.nativeReceipt(join(result.journalPath, 'result.json'), result);
    if (bound.nativeReceipt !== null) {
      if (localRecordHash(bound.nativeReceipt) !== localRecordHash(nativeReceipt))
        throw Error('root_initialization_receipt_invalid');
      return structuredClone(bound);
    }
    registry = parseLocalRegistry(await this.registry.load());
    const current = registry.operations.find((o) => o.actionId === bound.actionId);
    if (!current || localRecordHash(current) !== localRecordHash(bound))
      throw Error('operation_conflict');
    const applied =
      result.stage === 'applied' &&
      result.created === true &&
      result.quiescent &&
      result.stagingIdentity !== null;
    const completed: LocalRootInitializationOperation = {
      ...bound,
      stage: applied ? 'committed' : 'prepared',
      nativeReceipt,
    };
    await this.registry.compareAndSwap(registry.revision, {
      ...registry,
      revision: registry.revision + 1,
      roots: registry.roots.map((r) =>
        r.rootId === root.rootId && applied
          ? {
              ...r,
              staging: {
                path: join(root.path, '.agora-operations'),
                identity: result.stagingIdentity as string,
              },
            }
          : r,
      ),
      operations: registry.operations.map((o) => (o.actionId === bound.actionId ? completed : o)),
    });
    return completed;
  }
}
