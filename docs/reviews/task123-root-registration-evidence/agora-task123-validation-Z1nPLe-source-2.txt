/** Internal trusted control-plane coordinator. Never bind this API directly to
 * model/HTTP input: the caller must validate the Leader proposal and root selection.
 * This closes canonical references; it does not create filesystem capabilities.
 */
import {
  type AppState,
  applyMutations,
  assertLocalExecutionState,
  assertLocalExecutionTransition,
  type LocalExecutionV1,
  type Mutation,
} from '@agora/core-domain';
import { LocalRegistryFile, type LocalRegistryOwner } from './local-registry-file';
import {
  assertLocalRegistryTransition,
  isLocalBindingOperation,
  type LocalBindingOperation,
  type LocalRegistryRecords,
  type LocalRootInitializationOperation,
  localRecordHash,
  parseLocalRegistry,
} from './local-registry-records';

type Scope = { projectId: string; taskId: string };
interface CanonicalTasks {
  load(scope: Scope): Promise<AppState | undefined>;
  commit(
    scope: Scope,
    mutations: readonly Mutation[],
  ): Promise<{ state: AppState; changed: boolean }>;
}
export type LocalBindingRequest = Scope & {
  actionId: string;
  sourceMessageId: string;
  expectedRevision: number;
  /** No receipt for this action yet; the coordinator creates its canonical reference. */
  nextLocalExecution: LocalExecutionV1;
  records: Pick<LocalRegistryRecords, 'roots' | 'grants' | 'workspaces' | 'claims'>;
};
const requestKeys = [
  'projectId',
  'taskId',
  'actionId',
  'sourceMessageId',
  'expectedRevision',
  'nextLocalExecution',
  'records',
];
const id = (s: string) => /^[A-Za-z0-9][A-Za-z0-9._:-]{0,127}$/.test(s);
const localHash = (state: AppState) => localRecordHash(state.localExecution ?? null);
function leader(state: AppState, msgId: string) {
  const messages = state.messages.filter((m) => m.msgId === msgId);
  if (
    messages.length !== 1 ||
    messages[0]?.fromRole !== 'leader' ||
    messages[0]?.channelId !== 'main'
  )
    throw new Error('workspace_control_source_invalid');
}
function canonicalTask(state: AppState | undefined, scope: Scope): AppState {
  if (!state || state.projectId !== scope.projectId || state.taskId !== scope.taskId)
    throw new Error('workspace_task_scope_mismatch');
  assertLocalExecutionState(state);
  return state;
}
function receipt(state: AppState, operation: LocalBindingOperation) {
  return state.localExecution?.receipts.find(
    (r) =>
      r.receiptId === operation.receiptId &&
      r.actionId === operation.actionId &&
      r.inputHash === operation.inputHash &&
      r.registryRevision === operation.preparedRevision,
  );
}

export class LocalBindingCoordinator {
  private tail: Promise<unknown> = Promise.resolve();
  private constructor(
    private readonly registry: LocalRegistryFile,
    private readonly tasks: CanonicalTasks,
  ) {}
  static async open(owner: LocalRegistryOwner, tasks: CanonicalTasks, initialize = false) {
    return new LocalBindingCoordinator(
      await LocalRegistryFile.open(
        owner,
        parseLocalRegistry,
        initialize,
        assertLocalRegistryTransition,
        isLocalBindingOperation,
      ),
      tasks,
    );
  }
  private serial<T>(fn: () => Promise<T>): Promise<T> {
    const run = this.tail.catch(() => undefined).then(fn);
    this.tail = run.then(
      () => undefined,
      () => undefined,
    );
    return run;
  }
  async snapshot() {
    return parseLocalRegistry(await this.registry.load());
  }

  commitBinding(input: LocalBindingRequest): Promise<LocalBindingOperation> {
    // Validate JSON before cloning, and snapshot before the first await.
    const inputHash = localRecordHash(input);
    if (
      Object.keys(input).length !== requestKeys.length ||
      !requestKeys.every((k) => Object.hasOwn(input, k)) ||
      ![input.projectId, input.taskId, input.actionId, input.sourceMessageId].every(
        (x) => typeof x === 'string' && id(x),
      ) ||
      !Number.isSafeInteger(input.expectedRevision) ||
      input.expectedRevision < 0 ||
      Object.keys(input.records).sort().join(',') !== 'claims,grants,roots,workspaces'
    )
      return Promise.reject(new Error('workspace_control_input_invalid'));
    const request = structuredClone(input);
    return this.serial(async () => {
      const current = await this.snapshot();
      const existing = current.operations.find((o) => o.actionId === request.actionId);
      if (existing) {
        if (!isLocalBindingOperation(existing)) throw new Error('operation_conflict');
        if (
          existing.inputHash !== inputHash ||
          existing.projectId !== request.projectId ||
          existing.taskId !== request.taskId
        )
          throw new Error('operation_conflict');
        return this.finish(existing);
      }
      if (current.operations.some((o) => o.stage === 'prepared'))
        throw new Error('registry_recovery_required');
      if (
        current.revision !== request.expectedRevision ||
        !Number.isSafeInteger(current.revision + 2)
      )
        throw new Error('registry_revision_conflict');
      const state = canonicalTask(await this.tasks.load(request), request);
      leader(state, request.sourceMessageId);
      const preparedRevision = current.revision + 1;
      const receiptId = `binding:${request.actionId}`;
      if (receiptId.length > 128) throw new Error('workspace_control_input_invalid');
      const nextLocalExecution = structuredClone(request.nextLocalExecution);
      nextLocalExecution.receipts.push({
        receiptId,
        actionId: request.actionId,
        inputHash,
        registryRevision: preparedRevision,
      });
      applyMutations(state, [{ op: 'set', field: 'localExecution', value: nextLocalExecution }]);
      const operation: LocalBindingOperation = {
        actionId: request.actionId,
        inputHash,
        receiptId,
        projectId: request.projectId,
        taskId: request.taskId,
        preparedRevision,
        stage: 'prepared',
        previousLocalHash: localHash(state),
        nextLocalExecution,
        sourceMessageId: request.sourceMessageId,
      };
      const next = parseLocalRegistry({
        ...current,
        ...request.records,
        revision: preparedRevision,
        operations: [...current.operations, operation],
      });
      // A control request cannot alter another project's capability records.
      for (const name of ['roots', 'grants', 'workspaces', 'claims'] as const) {
        const others = (items: LocalRegistryRecords[typeof name]) =>
          items.filter((v) => v.projectId !== request.projectId);
        if (localRecordHash(others(current[name])) !== localRecordHash(others(next[name])))
          throw new Error('workspace_control_scope_mismatch');
      }
      await this.registry.compareAndSwap(current.revision, next);
      return this.finish(operation);
    });
  }

  recover(actionId: string, inputHash: string): Promise<LocalBindingOperation> {
    return this.serial(async () => {
      const snapshot = await this.snapshot();
      const operation = snapshot.operations.find((o) => o.actionId === actionId);
      if (!operation || !isLocalBindingOperation(operation) || operation.inputHash !== inputHash)
        throw new Error('operation_conflict');
      return this.finish(operation);
    });
  }

  /** Admission rechecks BOTH stores, including every referenced operation. */
  assertClosed(scope: Scope): Promise<AppState> {
    return this.readClosed(scope);
  }

  /** Internal initializer may inspect its already-committed grant source while
   * its exact prepared technical operation blocks ordinary workspace admission. */
  assertInitializationSource(operation: LocalRootInitializationOperation): Promise<AppState> {
    return this.readClosed(operation, operation);
  }

  private async readClosed(
    scope: Scope,
    initialization?: LocalRootInitializationOperation,
  ): Promise<AppState> {
    const registry = await this.snapshot();
    if (
      initialization &&
      (initialization.stage !== 'prepared' ||
        !registry.operations.some(
          (o) =>
            !isLocalBindingOperation(o) && localRecordHash(o) === localRecordHash(initialization),
        ))
    )
      throw new Error('workspace_initialization_source_invalid');
    if (
      registry.operations.some(
        (o) => o.stage === 'prepared' && o.actionId !== initialization?.actionId,
      )
    )
      throw new Error('registry_recovery_required');
    const state = canonicalTask(await this.tasks.load(scope), scope);
    if (!state.localExecution) throw new Error('workspace_binding_missing');
    for (const ref of state.localExecution.receipts) {
      const op = registry.operations
        .filter(isLocalBindingOperation)
        .find(
          (o) =>
            o.receiptId === ref.receiptId &&
            o.actionId === ref.actionId &&
            o.inputHash === ref.inputHash &&
            o.preparedRevision === ref.registryRevision &&
            o.projectId === scope.projectId &&
            o.taskId === scope.taskId &&
            o.stage === 'committed',
        );
      if (!op) throw new Error('workspace_binding_incomplete');
      leader(state, op.sourceMessageId);
    }
    if (!state.localExecution.receipts.length) throw new Error('workspace_binding_incomplete');
    const latestOperation = registry.operations
      .filter(isLocalBindingOperation)
      .filter((o) => o.projectId === scope.projectId && o.taskId === scope.taskId)
      .sort((a, b) => b.preparedRevision - a.preparedRevision)[0];
    if (
      !latestOperation ||
      localRecordHash(latestOperation.nextLocalExecution) !== localHash(state)
    )
      throw new Error('workspace_binding_incomplete');
    // Re-read after the task read: concurrent registry mutation invalidates admission.
    if ((await this.snapshot()).revision !== registry.revision)
      throw new Error('registry_revision_conflict');
    return state;
  }

  private async finish(operation: LocalBindingOperation): Promise<LocalBindingOperation> {
    const before = await this.snapshot();
    const current = before.operations.find((o) => o.actionId === operation.actionId);
    if (
      !current ||
      !isLocalBindingOperation(current) ||
      localRecordHash({ ...current, stage: operation.stage }) !== localRecordHash(operation)
    )
      throw new Error('operation_conflict');
    let state = canonicalTask(await this.tasks.load(operation), operation);
    leader(state, operation.sourceMessageId);
    if (current.stage === 'committed') {
      await this.assertClosed(operation);
      if (!receipt(state, current)) throw new Error('workspace_binding_incomplete');
      // Superseding bindings retain every immutable fact of this committed action.
      assertLocalExecutionTransition(current.nextLocalExecution, state.localExecution);
      return current;
    }
    if (localHash(state) !== localRecordHash(operation.nextLocalExecution)) {
      if (receipt(state, operation) || localHash(state) !== operation.previousLocalHash)
        throw new Error('workspace_binding_conflict');
      await this.tasks.commit(operation, [
        { op: 'set', field: 'localExecution', value: operation.nextLocalExecution },
      ]);
      state = canonicalTask(await this.tasks.load(operation), operation);
    }
    if (
      !receipt(state, operation) ||
      localHash(state) !== localRecordHash(operation.nextLocalExecution)
    )
      throw new Error('workspace_binding_incomplete');
    leader(state, operation.sourceMessageId);
    const latest = await this.snapshot();
    if (latest.revision !== before.revision) throw new Error('registry_revision_conflict');
    const committed = { ...operation, stage: 'committed' as const };
    await this.registry.compareAndSwap(latest.revision, {
      ...latest,
      revision: latest.revision + 1,
      operations: latest.operations.map((o) => (o.actionId === operation.actionId ? committed : o)),
    });
    return committed;
  }
}
