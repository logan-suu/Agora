/** Strict data validation only; registration does not create a live capability. */
import { createHash } from 'node:crypto';
import { dirname, isAbsolute, normalize } from 'node:path';
import {
  assertLocalExecutionTransition,
  assertWorkspaceRefsTransition,
  isLocalExecutionV1,
  isWorkspaceRefsV1,
  type LocalExecutionV1,
  type WorkspaceRefV1,
} from '@agora/core-domain';

export interface LocalRootRecord {
  rootId: string;
  projectId: string;
  selectionRef: string;
  path: string;
  volumeId: string;
  dev: string;
  inode: string;
  chain: { path: string; identity: string }[];
  staging: { path: string; identity: string } | null;
  inspectionHash: string;
}
export interface LocalGrantRecord {
  grantId: string;
  projectId: string;
  rootId: string;
  revision: number;
  policyVersion: string;
  actions: ('read' | 'edit' | 'run' | 'install' | 'generate' | 'remove')[];
  toolchainHash: string;
  networkHash: string;
  policyHash: string;
  createdActionId: string;
  leaderMessageId: string;
  status: 'active' | 'revoking' | 'revoked';
  revocationActionId: string | null;
}
export interface LocalClaimRecord {
  claimId: string;
  projectId: string;
  taskId: string;
  workspaceId: string;
  workerId: string;
  writerEpoch: number;
  createdActionId: string;
  status: 'active' | 'draining' | 'quarantined' | 'released';
  closureReceiptId: string | null;
}
export interface LocalBindingOperation {
  actionId: string;
  inputHash: string;
  receiptId: string;
  projectId: string;
  taskId: string;
  preparedRevision: number;
  stage: 'prepared' | 'committed';
  previousLocalHash: string;
  nextLocalExecution: LocalExecutionV1;
  sourceMessageId: string;
}
export interface LocalRootInitializationOperation {
  kind: 'root-initialization';
  actionId: string;
  inputHash: string;
  receiptId: string;
  projectId: string;
  taskId: string;
  rootId: string;
  grantId: string;
  grantRevision: number;
  sourceMessageId: string;
  preparedRevision: number;
  stage: 'prepared' | 'committed';
  nativeReceipt: null | {
    sha256: string;
    stage: 'applied' | 'conflict' | 'recoveryRequired';
    created: boolean | null;
    stagingIdentity: string | null;
    quiescent: boolean;
  };
}
export type LocalRegistryOperation = LocalBindingOperation | LocalRootInitializationOperation;
export function isLocalBindingOperation(
  value: LocalRegistryOperation,
): value is LocalBindingOperation {
  return !('kind' in value);
}
export interface LocalRegistryRecords {
  schemaVersion: 'local-workspaces-v1';
  revision: number;
  roots: LocalRootRecord[];
  grants: LocalGrantRecord[];
  workspaces: WorkspaceRefV1[];
  claims: LocalClaimRecord[];
  operations: LocalRegistryOperation[];
}
function fail(): never {
  throw new Error('invalid_local_registry_records');
}
const id = (v: unknown): v is string =>
  typeof v === 'string' && /^[A-Za-z0-9][A-Za-z0-9._:-]{0,127}$/.test(v);
const hash = (v: unknown): v is string => typeof v === 'string' && /^[a-f0-9]{64}$/.test(v);
const integer = (v: unknown): v is number =>
  typeof v === 'number' && Number.isSafeInteger(v) && v >= 0;
const decimal = (v: unknown): v is string =>
  typeof v === 'string' && /^(0|[1-9][0-9]{0,19})$/.test(v);
const identity = (v: unknown): v is string =>
  typeof v === 'string' && v.split(':').length === 2 && v.split(':').every(decimal);
const path = (v: unknown): v is string =>
  typeof v === 'string' &&
  Buffer.byteLength(v) <= 4096 &&
  isAbsolute(v) &&
  normalize(v) === v &&
  ![...v].some((c) => c.charCodeAt(0) < 32 || c.charCodeAt(0) === 127);
function object(v: unknown, keys: string[]): Record<string, unknown> {
  if (
    !v ||
    typeof v !== 'object' ||
    Array.isArray(v) ||
    Reflect.ownKeys(v).length !== keys.length ||
    !keys.every((k) => Object.hasOwn(v, k))
  )
    fail();
  return v as Record<string, unknown>;
}
/** Bounded JSON traversal rejects accessors, prototypes, cycles and hidden data. */
function json(v: unknown, depth = 0, ancestors = new Set<object>()): void {
  if (depth > 32) fail();
  if (
    v === null ||
    typeof v === 'string' ||
    typeof v === 'boolean' ||
    (typeof v === 'number' && Number.isFinite(v))
  )
    return;
  if (typeof v !== 'object' || ancestors.has(v)) fail();
  const prototype = Object.getPrototypeOf(v);
  if (
    Array.isArray(v)
      ? prototype !== Array.prototype || v.length > 4096
      : prototype !== null && prototype !== Object.prototype
  )
    fail();
  ancestors.add(v);
  const keys = Reflect.ownKeys(v);
  if (Array.isArray(v) && keys.length !== v.length + 1) fail();
  for (const k of keys) {
    if (Array.isArray(v) && k === 'length') continue;
    const d = Object.getOwnPropertyDescriptor(v, k);
    if (typeof k !== 'string' || !d?.enumerable || !Object.hasOwn(d, 'value')) fail();
    json(d.value, depth + 1, ancestors);
  }
  if (
    Array.isArray(v) &&
    Array.from({ length: v.length }, (_, i) => String(i)).some((k) => !Object.hasOwn(v, k))
  )
    fail();
  ancestors.delete(v);
}
function canonical(v: unknown): string {
  if (v === null || typeof v !== 'object') return JSON.stringify(v);
  if (Array.isArray(v)) return `[${v.map(canonical).join(',')}]`;
  return `{${Object.keys(v)
    .sort()
    .map((k) => `${JSON.stringify(k)}:${canonical((v as Record<string, unknown>)[k])}`)
    .join(',')}}`;
}
export function localRecordHash(v: unknown): string {
  json(v);
  return createHash('sha256').update(canonical(v)).digest('hex');
}
function array(v: unknown): unknown[] {
  if (!Array.isArray(v) || v.length > 4096) fail();
  return v;
}
function unique<T>(items: T[], key: (v: T) => unknown): void {
  if (new Set(items.map(key)).size !== items.length) fail();
}
function rootRecord(v: unknown): LocalRootRecord {
  const r = object(v, [
    'rootId',
    'projectId',
    'selectionRef',
    'path',
    'volumeId',
    'dev',
    'inode',
    'chain',
    'staging',
    'inspectionHash',
  ]);
  if (
    !['rootId', 'projectId', 'selectionRef', 'volumeId'].every((k) => id(r[k])) ||
    !path(r.path) ||
    r.path === '/' ||
    !decimal(r.dev) ||
    !decimal(r.inode) ||
    !hash(r.inspectionHash)
  )
    fail();
  const chain = array(r.chain);
  if (!chain.length || chain.length > 128) fail();
  for (let i = 0; i < chain.length; i++) {
    const e = object(chain[i], ['path', 'identity']);
    if (
      !path(e.path) ||
      !identity(e.identity) ||
      (i === 0 ? e.path !== '/' : dirname(e.path) !== (chain[i - 1] as { path: string }).path)
    )
      fail();
  }
  const last = chain.at(-1) as { path: string; identity: string };
  const staging = r.staging === null ? null : object(r.staging, ['path', 'identity']);
  if (
    last.path !== r.path ||
    last.identity !== `${r.dev}:${r.inode}` ||
    (staging !== null &&
      (!path(staging.path) ||
        dirname(staging.path) !== r.path ||
        !identity(staging.identity) ||
        staging.identity.split(':')[0] !== r.dev))
  )
    fail();
  return v as LocalRootRecord;
}
function grantRecord(v: unknown): LocalGrantRecord {
  const r = object(v, [
    'grantId',
    'projectId',
    'rootId',
    'revision',
    'policyVersion',
    'actions',
    'toolchainHash',
    'networkHash',
    'policyHash',
    'createdActionId',
    'leaderMessageId',
    'status',
    'revocationActionId',
  ]);
  if (
    ![
      'grantId',
      'projectId',
      'rootId',
      'policyVersion',
      'createdActionId',
      'leaderMessageId',
    ].every((k) => id(r[k])) ||
    !['toolchainHash', 'networkHash', 'policyHash'].every((k) => hash(r[k])) ||
    !integer(r.revision) ||
    !['active', 'revoking', 'revoked'].includes(r.status as string)
  )
    fail();
  const actions = array(r.actions);
  if (
    !actions.length ||
    actions.some(
      (a) => !['read', 'edit', 'run', 'install', 'generate', 'remove'].includes(a as string),
    )
  )
    fail();
  unique(actions, (a) => a);
  if (r.status === 'active' ? r.revocationActionId !== null : !id(r.revocationActionId)) fail();
  return v as LocalGrantRecord;
}
function claimRecord(v: unknown): LocalClaimRecord {
  const r = object(v, [
    'claimId',
    'projectId',
    'taskId',
    'workspaceId',
    'workerId',
    'writerEpoch',
    'createdActionId',
    'status',
    'closureReceiptId',
  ]);
  if (
    !['claimId', 'projectId', 'taskId', 'workspaceId', 'workerId', 'createdActionId'].every((k) =>
      id(r[k]),
    ) ||
    !integer(r.writerEpoch) ||
    !['active', 'draining', 'quarantined', 'released'].includes(r.status as string) ||
    (r.status === 'released' ? !id(r.closureReceiptId) : r.closureReceiptId !== null)
  )
    fail();
  return v as LocalClaimRecord;
}
function bindingOperationRecord(v: unknown): LocalBindingOperation {
  const r = object(v, [
    'actionId',
    'inputHash',
    'receiptId',
    'projectId',
    'taskId',
    'preparedRevision',
    'stage',
    'previousLocalHash',
    'nextLocalExecution',
    'sourceMessageId',
  ]);
  if (
    !['actionId', 'receiptId', 'projectId', 'taskId', 'sourceMessageId'].every((k) => id(r[k])) ||
    !hash(r.inputHash) ||
    !hash(r.previousLocalHash) ||
    !integer(r.preparedRevision) ||
    r.preparedRevision === 0 ||
    !['prepared', 'committed'].includes(r.stage as string) ||
    !isLocalExecutionV1(r.nextLocalExecution)
  )
    fail();
  const ref = r.nextLocalExecution.receipts.find((x) => x.receiptId === r.receiptId);
  if (
    !ref ||
    ref.actionId !== r.actionId ||
    ref.inputHash !== r.inputHash ||
    ref.registryRevision !== r.preparedRevision ||
    r.nextLocalExecution.workspaces.some(
      (w) => w.projectId !== r.projectId || w.taskId !== r.taskId,
    )
  )
    fail();
  return v as LocalBindingOperation;
}

function operationRecord(v: unknown): LocalRegistryOperation {
  if (!v || typeof v !== 'object' || !Object.hasOwn(v, 'kind')) return bindingOperationRecord(v);
  const r = object(v, [
    'kind',
    'actionId',
    'inputHash',
    'receiptId',
    'projectId',
    'taskId',
    'rootId',
    'grantId',
    'grantRevision',
    'sourceMessageId',
    'preparedRevision',
    'stage',
    'nativeReceipt',
  ]);
  if (
    r.kind !== 'root-initialization' ||
    !['actionId', 'receiptId', 'projectId', 'taskId', 'rootId', 'grantId', 'sourceMessageId'].every(
      (k) => id(r[k]),
    ) ||
    !hash(r.inputHash) ||
    !integer(r.grantRevision) ||
    !integer(r.preparedRevision) ||
    r.preparedRevision === 0 ||
    !['prepared', 'committed'].includes(r.stage as string)
  )
    fail();
  if (r.nativeReceipt !== null) {
    const n = object(r.nativeReceipt, [
      'sha256',
      'stage',
      'created',
      'stagingIdentity',
      'quiescent',
    ]);
    if (
      !hash(n.sha256) ||
      !['applied', 'conflict', 'recoveryRequired'].includes(n.stage as string) ||
      !(n.created === null || typeof n.created === 'boolean') ||
      !(n.stagingIdentity === null || identity(n.stagingIdentity)) ||
      typeof n.quiescent !== 'boolean' ||
      (n.stage === 'applied' &&
        (n.created !== true || n.stagingIdentity === null || n.quiescent !== true))
    )
      fail();
  }
  if (
    r.stage === 'committed' &&
    (r.nativeReceipt === null || (r.nativeReceipt as { stage: string }).stage !== 'applied')
  )
    fail();
  return v as LocalRootInitializationOperation;
}

export function parseLocalRegistry(value: unknown): LocalRegistryRecords {
  json(value);
  const r = object(value, [
    'schemaVersion',
    'revision',
    'roots',
    'grants',
    'workspaces',
    'claims',
    'operations',
  ]);
  if (
    r.schemaVersion !== 'local-workspaces-v1' ||
    !integer(r.revision) ||
    !isWorkspaceRefsV1(r.workspaces)
  )
    fail();
  const workspaces = r.workspaces;
  const roots = array(r.roots).map(rootRecord),
    grants = array(r.grants).map(grantRecord),
    claims = array(r.claims).map(claimRecord),
    operations = array(r.operations).map(operationRecord);
  unique(roots, (x) => x.rootId);
  unique(roots, (x) => `${x.dev}:${x.inode}`);
  unique(roots, (x) => x.path);
  unique(grants, (x) => x.grantId);
  unique(claims, (x) => x.claimId);
  unique(operations, (x) => x.actionId);
  unique(operations, (x) => x.receiptId);
  unique(operations, (x) => x.preparedRevision);
  for (const g of grants)
    if (!roots.some((root) => root.rootId === g.rootId && root.projectId === g.projectId)) fail();
  for (const w of r.workspaces)
    if (
      !roots.some((root) => root.rootId === w.rootId && root.staging !== null) ||
      !grants.some(
        (g) => g.grantId === w.grantId && g.rootId === w.rootId && g.projectId === w.projectId,
      )
    )
      fail();
  for (const c of claims)
    if (
      !workspaces.some(
        (w) =>
          w.workspaceId === c.workspaceId && w.projectId === c.projectId && w.taskId === c.taskId,
      )
    )
      fail();
  const occupied = claims.filter((c) => c.status !== 'released');
  for (let i = 0; i < occupied.length; i++)
    for (let j = 0; j < i; j++) {
      const a = roots.find((root) =>
        workspaces.some(
          (w) => w.workspaceId === occupied[i]?.workspaceId && w.rootId === root.rootId,
        ),
      );
      const b = roots.find((root) =>
        workspaces.some(
          (w) => w.workspaceId === occupied[j]?.workspaceId && w.rootId === root.rootId,
        ),
      );
      if (
        !a ||
        !b ||
        a.chain.some((e) => e.identity === `${b.dev}:${b.inode}`) ||
        b.chain.some((e) => e.identity === `${a.dev}:${a.inode}`)
      )
        fail();
    }
  if (operations.filter((o) => o.stage === 'prepared').length > 1) fail();
  unique(
    operations.filter((o): o is LocalRootInitializationOperation => !isLocalBindingOperation(o)),
    (o) => o.rootId,
  );
  for (const o of operations) {
    if (!isLocalBindingOperation(o)) {
      const root = roots.find((r) => r.rootId === o.rootId && r.projectId === o.projectId);
      const grant = grants.find(
        (g) => g.grantId === o.grantId && g.rootId === o.rootId && g.projectId === o.projectId,
      );
      if (
        !root ||
        !grant ||
        grant.revision < o.grantRevision ||
        o.preparedRevision > r.revision ||
        (o.stage === 'committed' &&
          (!root.staging || root.staging.identity !== o.nativeReceipt?.stagingIdentity)) ||
        (o.stage === 'prepared' && root.staging !== null)
      )
        fail();
      continue;
    }
    if (
      o.preparedRevision > r.revision ||
      o.nextLocalExecution.rootIds.some(
        (id) => !roots.some((root) => root.rootId === id && root.projectId === o.projectId),
      )
    )
      fail();
    for (const w of o.nextLocalExecution.workspaces)
      if (!workspaces.some((current) => localRecordHash(w) === localRecordHash(current))) fail();
  }
  for (const root of roots)
    if (
      root.staging !== null &&
      !operations.some(
        (o) =>
          !isLocalBindingOperation(o) &&
          o.rootId === root.rootId &&
          o.stage === 'committed' &&
          o.nativeReceipt?.stagingIdentity === root.staging?.identity,
      )
    )
      fail();
  return structuredClone(value) as LocalRegistryRecords;
}

/** Executed under the file CAS lock, never solely against a stale caller snapshot. */
export function assertLocalRegistryTransition(previous: unknown, next: unknown): void {
  const before = parseLocalRegistry(previous),
    after = parseLocalRegistry(next);
  if (after.revision !== before.revision + 1 || !Number.isSafeInteger(after.revision)) fail();
  assertWorkspaceRefsTransition(before.workspaces, after.workspaces);
  for (const root of after.roots) {
    const old = before.roots.find((r) => r.rootId === root.rootId);
    if (!old) {
      if (root.staging !== null) fail();
      continue;
    }
    const { staging: previousStaging, ...fixed } = old;
    const { staging: nextStaging, ...nextFixed } = root;
    if (localRecordHash(fixed) !== localRecordHash(nextFixed)) fail();
    if (localRecordHash(previousStaging) === localRecordHash(nextStaging)) continue;
    if (
      previousStaging !== null ||
      nextStaging === null ||
      !after.operations.some(
        (op) =>
          !isLocalBindingOperation(op) &&
          op.rootId === root.rootId &&
          op.stage === 'committed' &&
          op.nativeReceipt?.stagingIdentity === nextStaging.identity &&
          before.operations.some(
            (prior) =>
              !isLocalBindingOperation(prior) &&
              prior.actionId === op.actionId &&
              prior.stage === 'prepared',
          ),
      )
    )
      fail();
  }
  for (const root of before.roots) if (!after.roots.some((r) => r.rootId === root.rootId)) fail();
  for (const g of before.grants) {
    const n = after.grants.find((x) => x.grantId === g.grantId);
    if (!n) fail();
    const { status: _s, revision: _r, revocationActionId: _a, ...fixed } = g;
    const { status: _ns, revision: _nr, revocationActionId: _na, ...nextFixed } = n;
    if (localRecordHash(fixed) !== localRecordHash(nextFixed)) fail();
    if (localRecordHash(g) === localRecordHash(n)) continue;
    if (
      n.revision !== g.revision + 1 ||
      g.status === 'revoked' ||
      n.status === 'active' ||
      (g.revocationActionId !== null && g.revocationActionId !== n.revocationActionId)
    )
      fail();
  }
  for (const c of before.claims) {
    const n = after.claims.find((x) => x.claimId === c.claimId);
    if (!n) fail();
    const { status: _s, closureReceiptId: _r, ...fixed } = c;
    const { status: _ns, closureReceiptId: _nr, ...nextFixed } = n;
    if (localRecordHash(fixed) !== localRecordHash(nextFixed)) fail();
    if (localRecordHash(c) === localRecordHash(n)) continue;
    if (
      c.status === 'released' ||
      n.status === 'active' ||
      (c.status === 'quarantined' && n.status !== 'released')
    )
      fail();
  }
  for (const o of before.operations) {
    const n = after.operations.find((x) => x.actionId === o.actionId);
    if (!n || isLocalBindingOperation(o) !== isLocalBindingOperation(n)) fail();
    if (isLocalBindingOperation(o) && isLocalBindingOperation(n)) {
      const { stage: _s, ...fixed } = o,
        { stage: _ns, ...nextFixed } = n;
      if (
        localRecordHash(fixed) !== localRecordHash(nextFixed) ||
        (o.stage === 'committed' && n.stage !== 'committed')
      )
        fail();
      assertLocalExecutionTransition(o.nextLocalExecution, n.nextLocalExecution);
    } else if (!isLocalBindingOperation(o) && !isLocalBindingOperation(n)) {
      const { stage: _s, nativeReceipt: previousReceipt, ...fixed } = o;
      const { stage: _ns, nativeReceipt: nextReceipt, ...nextFixed } = n;
      if (
        localRecordHash(fixed) !== localRecordHash(nextFixed) ||
        (o.stage === 'committed' && n.stage !== 'committed') ||
        (previousReceipt !== null &&
          localRecordHash(previousReceipt) !== localRecordHash(nextReceipt))
      )
        fail();
    }
  }
  for (const o of after.operations.filter(
    (n) => !before.operations.some((o) => o.actionId === n.actionId),
  ))
    if (
      o.stage !== 'prepared' ||
      o.preparedRevision !== after.revision ||
      (!isLocalBindingOperation(o) && o.nativeReceipt !== null)
    )
      fail();
}
