// Real files, native processes and Seatbelt. A fixed trusted authority snapshot
// exercises the registry seam; it is not a substitute for the pending product registry.
import { expect, it } from 'vitest';
import { probeLocalCommandBinding } from './local-command-binding-fixture';

for (const scenario of ['grant-revision', 'worker', 'writer-epoch', 'authority-error'] as const) {
  it(`stops registered execution when ${scenario} changes`, async () => {
    const result = await probeLocalCommandBinding(scenario);
    expect(result.result.released).toBe(true);
    expect(result.result.error).toBe('command_binding_invalidated');
    expect(result.result.observation?.bindingFailure).toBe('authority_changed');
    expect(result.result.observation?.stop.registeredState).toBe('stopped');
    expect(result.record.binding).not.toBeNull();
    expect(result.record.resourceState).toBe('quarantined');
  }, 20_000);
}

it('refuses to release project code after a root is replaced', async () => {
  const result = await probeLocalCommandBinding('root-before-release');
  expect(result.result.released).toBe(false);
  expect(result.marker).toBe(false);
  expect(result.bindingFailure).toBe('root_changed');
}, 20_000);

it('stops on ancestor relocation and never follows the moved directory', async () => {
  const result = await probeLocalCommandBinding('ancestor');
  expect(result.result.released).toBe(true);
  expect(result.result.observation?.bindingFailure).toBe('root_changed');
  expect(result.result.observation?.stop.registeredState).toBe('stopped');
  expect(result.replacementUnchanged).toBe(true);
}, 20_000);

it('rejects tool replacement before any process is launched', async () => {
  const result = await probeLocalCommandBinding('tool-before-admission');
  expect(result.result.released).toBe(false);
  expect(result.record.birthIdentities).toHaveLength(0);
  expect(result.bindingFailure).toBe('tool_changed');
}, 20_000);

it('keeps invalidation sticky even when the authority is restored', async () => {
  const result = await probeLocalCommandBinding('sticky');
  expect(result.stickyChecks).toEqual(['authority_changed', 'authority_changed']);
  expect(result.result.released).toBe(false);
}, 20_000);

it('rejects command arguments that differ from the captured binding', async () => {
  const result = await probeLocalCommandBinding('arguments');
  expect(result.result.released).toBe(false);
  expect(result.record.birthIdentities).toHaveLength(0);
}, 20_000);

it('rechecks authority at completion instead of accepting a successful process exit', async () => {
  const result = await probeLocalCommandBinding('completion');
  expect(result.result.released).toBe(true);
  expect(result.result.observation?.bindingFailure).toBe('authority_changed');
  expect(result.result.error).toBe('command_binding_invalidated');
  expect(result.result.payloadResult).toEqual({ exitCode: 0, signal: null });
}, 20_000);

it('never executes a replaced process-control helper even during cleanup', async () => {
  const result = await probeLocalCommandBinding('control-drift');
  expect(result.result.released).toBe(true);
  expect(result.result.observation?.bindingFailure).toBe('tool_changed');
  expect(result.result.observation?.stop.registeredState).toBe('needsAttention');
  expect(result.result.observation?.stop.signals).toEqual([]);
  expect(result.trapExecuted).toBe(false);
  expect(result.fixtureStop?.registeredState).toBe('stopped');
}, 20_000);

it('rejects old journal formats and malformed persisted root bindings', async () => {
  const result = await probeLocalCommandBinding('binding-journal');
  expect(result.journalErrors).toEqual(['invalid_command_journal', 'invalid_command_journal']);
  expect(result.record.binding).not.toBeNull();
}, 20_000);
